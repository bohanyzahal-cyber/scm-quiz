# title: השלמות מסיכומי הכיתה — מה שנאמר בשיעורים
# פרק השלמה (3.11.2026). מבוסס על שני סיכומי השיעורים של הכיתה (ניצן — כמותי, שלמה — תיאורטי), ועל 28 השאלות שנוספו מהם לאתר (בנק 26).
# רק מה שלא היה בפרקים הקודמים. ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק הַשְׁלָמָה. חֲבֵרִים מֵהַכִּתָה שָׁלְחוּ לָנוּ שְׁנֵי סִכּוּמִים שֶׁל כֹּל הַשִׁעוּרִים, אֶחָד שֶׁל נִצָן וְאֶחָד שֶׁל שְׁלֹמֹה. עָבַרְנוּ עֲלֵיהֶם מוּל אֲתַר הַתִרְגוּל. מָה מָצָאנוּ?

א: כִּמְעַט הַכֹּל כְּבָר הָיָה. אֲבָל הָיוּ כַּמָה דְבָרִים שֶׁנֶאֶמְרוּ בַּשִׁעוּרִים, וְלֹא הָיוּ עֲלֵיהֶם שְׁאֵלוֹת. הוֹסַפְנוּ עֶשְׂרִים וּשְׁמוֹנֶה שְׁאֵלוֹת, וּבַפֶּרֶק הַזֶה נַעֲבֹר עֲלֵיהֶם. נַתְחִיל בְּנִצָן.

ש: יַאלְלָה.

א: בְּשִׁעוּר עַל אַחְזָקָה, נִצָן הִגְדִיר תִפְעוּל בְּמִלָה אַחַת: הֲמָרָה. הֲמָרָה שֶׁל תְשׂוּמוֹת לַתְפוּקוֹת.
א: וְכֹל אִרְגוּן קָם בִּשְׁבִיל פְּעִילוּת מֶרְכָּזִית אַחַת, שֶׁהוּא קוֹרֵא לָה לִבַּת הָאִרְגוּן. בָּאוּנִיבֶרְסִיטָה, הַלִבָּה הִיא הוֹרָאָה וּמֶחְקָר, כְּלוֹמַר אַסְפָּקַת יֶדַע. וְהַלָקוֹחוֹת הֵם הַסְטוּדֶנְטִים וְהַמֶמְשָׁלָה.
א: הַתִפְעוּל מַזְרִים אֶל הַלִבָּה אַרְבָּעָה מַשְׁאַבִּים: כּוֹחַ אָדָם, חֹמֶר, יֶדַע וּכְסָפִים. וְחֹמֶר, אֶצְלוֹ, זֶה גַם מְלַאי וְגַם תַשְׁתִיוֹת יִצוּר, כְּמוֹ כִּתוֹת וְאוּלַמוֹת.

ש: וְאֵיפֹה הָאַחְזָקָה בַּתְמוּנָה הַזֹאת?

א: בְּתוֹךְ מַעֲרַךְ הַתִפְעוּל נִמְצֵאת הַלוֹגִיסְטִיקָה, וּבְתוֹךְ הַלוֹגִיסְטִיקָה נִמְצֵאת הָאַחְזָקָה. הַלוֹגִיסְטִיקָה אַחְרָאִית עַל זְרִימַת הַחֹמֶר, וְהָאַחְזָקָה אַחְרָאִית עַל כְּשִׁירוּת הַמַעֲרָכוֹת.

ש: וְהָיָה שָׁם גַם הֶבְדֵל בֵּין מוּצָר לְשֵׁרוּת.

א: נָכוֹן, שְׁלוֹשָׁה הֶבְדֵלִים. מוּחָשִׁיוּת: מוּצָר אֶפְשָׁר לְמַשֵׁשׁ, וְשֵׁרוּת לֹא. הַעֲבָרַת בַּעֲלוּת: כְּשֶׁקוֹנִים מוּצָר, הַבַּעֲלוּת עוֹבֶרֶת לַקוֹנֶה. בְּטִפּוּל רְפוּאִי אוֹ בְּשִׁעוּר, לֹא עוֹבֶרֶת בַּעֲלוּת עַל שׁוּם נֶכֶס. וְרַב גוֹנִיוּת: שֵׁרוּת מֻשְׁפָּע מֵהַרְבֵּה יוֹתֵר גוֹרְמִים מִמוּצָר.

ש: וּמֵאֵיפֹה בִּכְלָל הִתְחִילָה הַלוֹגִיסְטִיקָה?

א: לְפִי נִצָן, מִמִלְחֶמֶת הָעוֹלָם הָרִאשׁוֹנָה. לַצָבָא הַגֶרְמָנִי הָיָה צָבָא חָזָק, וְהוּא הִפְסִיד, כִּי לַחַיָלִים לֹא הָיוּ אֹכֶל וְאֶמְצְעֵי חִמוּם. מִשָׁם הִתְחִילוּ לְנַהֵל לוֹגִיסְטִיקָה, וּמִשָׁם זֶה עָבַר לָעוֹלָם הָאֶזְרָחִי.

ש: עַכְשָׁיו מְלַאי.

א: נִצָן מָנָה אַרְבָּעָה סוּגֵי מְלַאי בְּאִרְגוּן יַצְרָנִי: חוֹמְרֵי גֶלֶם, חֲלָקִים קְנוּיִים, תוֹצֶרֶת בְּתַהֲלִיךְ, וְתוֹצֶרֶת גְמוּרָה. הַחֲלָקִים הַקְנוּיִים הֵם רְכִיבִים שֶׁקוֹנִים מְסַפֵּק, וּמַרְכִּיבִים מֵהֶם מִכְלוֹלִים.
א: וְשִׂימוּ לֵב: שְׁלֹמֹה מוֹנֶה רַק שְׁלוֹשָׁה סוּגִים. חוֹמְרֵי גֶלֶם, תוֹצֶרֶת בְּתַהֲלִיךְ, וְתוֹצֶרֶת גְמוּרָה. בִּשְׁאֵלָה שֶׁל שְׁלֹמֹה, זוֹ הַתְשׁוּבָה.

ש: וּמָה עִם רֶכֶשׁ מִחוּץ לָאָרֶץ?

א: כְּשֶׁקוֹנִים לָרִאשׁוֹנָה מְסַפֵּק בְּחוּץ לָאָרֶץ שֶׁלֹא מַכִּירִים, יֵשׁ חֹסֶר אֵמוּן מִשְׁנֵי הַצְדָדִים. הַסַפָּק חוֹשֵׁשׁ שֶׁלֹא יְשַׁלְמוּ לוֹ, וְהַקוֹנֶה חוֹשֵׁשׁ לְשַׁלֵם וְלֹא לְקַבֵּל סְחוֹרָה.
א: הַפִּתְרוֹן שֶׁהֻזְכַּר הוּא מִכְתַב אַשְׁרַאי: הַבַּנְק מִתְחַיֵב לְשַׁלֵם, וְהַכֶּסֶף עוֹבֵר לְסַפֵּק רַק כְּשֶׁהוּא מַצִיג מִסְמָכִים שֶׁהַסְחוֹרָה נִשְׁלְחָה.

ש: וּבִמְלַאי בִּטָחוֹן?

א: נִצָן אָמַר שֶׁיֵשׁ פְּרִיטִים שֶׁלֹא מַחֲזִיקִים לָהֶם מְלַאי בִּטָחוֹן בִּכְלָל: מוּצָרִים שֶׁהַתֹקֶף שֶׁלָהֶם פָּג, פְּרִיטִים יִחוּדִיִים, וּפְרִיטִים יְקָרִים מְאוֹד, כְּמוֹ טַבְּעוֹת יַהֲלוֹם.

ש: וְאַחֲרוֹן אֵצֶל נִצָן, עוֹנָתִיוֹת.

א: כְּדֵי שֶׁמוּצָר יִהְיֶה עוֹנָתִי, צְרִיכִים לְהִתְקַיֵם שְׁנֵי דְבָרִים: בִּקוּשׁ שׁוֹנֶה בִּתְקוּפוֹת שׁוֹנוֹת, וּדְפוּס שֶׁחוֹזֵר עַל עַצְמוֹ בְּמַחֲזוֹרִיוּת.
א: וְזֶה לֹא חַיָב לִהְיוֹת קָשׁוּר לְעוֹנוֹת הַשָׁנָה. יַיִן לִפְנֵי פֶּסַח, צִיוּד מִשְׂרָדִי לִפְנֵי סֶפְּטֶמְבֶּר. וְהַמַחְזוֹר לֹא חַיָב לִהְיוֹת שָׁנָה. בְּדֻגְמָה שֶׁל הָרַכֶּבֶת הַקַלָה, הַמַחְזוֹר הוּא שָׁבוּעַ שֶׁל שִׁשָׁה יְמֵי עֲבוֹדָה.

ש: עוֹבְרִים לִשְׁלֹמֹה.

א: נַתְחִיל בְּמִקוּר חוּץ. כְּשֶׁיוֹצְאִים לְמִקוּר חוּץ, שְׁלֹמֹה מַמְלִיץ לִבְחֹר לִידִינְג פַּרְטְנֶר. סַפָּק מוֹבִיל, שֶׁיֵשׁ לוֹ יֶדַע, שֶׁמֵבִיא עֵרֶךְ, וְשֶׁלֹא צָרִיךְ לְלַמֵד אוֹתוֹ כֹּל פַּעַם מֵחָדָשׁ. בַּלָשׁוֹן שֶׁלוֹ: מִישֶׁהוּ שֶׁיִקַח אוֹתָנוּ קָדִימָה, וְלֹא יִמְשֹׁךְ אוֹתָנוּ אָחוֹרָה.
א: וּבַגִישָׁה הַגֵלוֹקָלִית: חוֹשְׁבִים גְלוֹבָּלִית וּפוֹעֲלִים מְקוֹמִית. בּוֹחֲרִים שֻׁתָף כָּזֶה, מַגְדִירִים לוֹ אֶת הַמְשִׂימָה, וּבוֹדְקִים אֶת הַתוֹצָאוֹת. אֶת הָאֵיךְ, מַשְׁאִירִים לוֹ.

ש: וּמָה זֶה שׁוֹנֶה מִקַבְּלָנוּת מִשְׁנֶה?

א: בְּקַבְּלָנוּת מִשְׁנֶה, אֲנַחְנוּ מְנַהֲלִים אֶת הַסַפָּק צַעַד אַחַר צַעַד. בְּמִקוּר חוּץ, מַעֲבִירִים לוֹ אֶת הָאַחְרָיוּת לַמְשִׂימָה כֻּלָה.

ש: עַכְשָׁיו נִהוּל חֹמֶר.

א: שְׁלוֹשָׁה דְבָרִים חֲדָשִׁים. הָרִאשׁוֹן, מְחִיקוֹת מְלַאי. מוֹחֲקִים מָנָה מֵהַמַעֲרֶכֶת, וְאָז מַפְנִים אוֹתָה לְהַשְׁמָדָה, לִתְרוּמָה אוֹ לַפַּח. וּשְׁלֹמֹה אָמַר: אִם יֵשׁ הַרְבֵּה הַשְׁמָדוֹת, זוֹ טָעוּת. זֶה סִימָן לְתִכְנוּן יֶתֶר, שֶׁהָפַךְ לַמְלַאי אִטִי, וְאַחַר כָּךְ לַמְלַאי מֵת.
א: וְיֵשׁ לָזֶה גַם צַד כַּסְפִּי. חֶבְרָה רִוְחִית עֲשׂוּיָה לִמְחֹק מְלַאי כְּדֵי לְהַקְטִין נְכָסִים, וּלְשַׁלֵם פָּחוֹת מַס.

ש: וְהַשֵׁנִי?

א: עֲלוּת חֹסֶן. שְׁלֹמֹה מָנָה שָׁלוֹשׁ עֲלוּיוֹת שֶׁצָרִיךְ לִשְׁקֹל בִּמְדִינִיוּת מְלַאי: עֲלוּת הוֹן, כִּי מְלַאי זֶה כֶּסֶף. פְּחָת. וַעֲלוּת חֹסֶן: מָה יִקְרֶה, וּמָה זֶה יַעֲלֶה, אִם יֶחְסַר פְּרִיט.
א: וּמִכָּאן הַמֶסֶר שֶׁלוֹ, שֶׁמְלַאי זֶה כּוֹחַ, וְאֵין לְפַחֵד מִמֶנוּ. הַחֲבָרוֹת הָרָזוֹת בְּיוֹתֵר נָפְלוּ כְּשֶׁחָסַר לָהֶן מְלַאי.

ש: וְהַשְׁלִישִׁי?

א: הַמְגֵרָה הַכְּפוּלָה. זוֹ שִׁיטָה לִפְרִיטֵי סִי, הַפְּרִיטִים הַזוֹלִים וְהַפָּחוֹת חֲשׁוּבִים. מַחֲזִיקִים אוֹתָם בִּשְׁתֵי מְגֵרוֹת. כְּשֶׁהָרִאשׁוֹנָה מִתְרוֹקֶנֶת, מַזְמִינִים, וּבֵינְתַיִם עוֹבְדִים מֵהַשְׁנִיָה. כָּךְ לֹא מַשְׁקִיעִים קֶשֶׁב נִהוּלִי בִּפְרִיטִים שֶׁעֶרְכָּם נָמוּךְ.

ש: וְהָיָה גַם מַשֶׁהוּ עַל חִזוּי.

א: כֵּן. נַנִיחַ שֶׁאַחֲרֵי שָׁנִים שֶׁל לְחִימָה, הַמְדִינָה עוֹבֶרֶת בְּבַת אַחַת לְשָׁלוֹם, וְצָרִיךְ לַחֲזוֹת כַּמָה יְמֵי מִלוּאִים יִדָרְשׁוּ. לְפִי שְׁלֹמֹה, נְתוּנֵי הֶעָבָר כְּבָר לֹא רֵלֵוַנְטִיִים. לֹא מְמֻצָע נָע, וְלֹא הַחְלָקָה מַעֲרִיכִית. כָּאן עָדִיף חִזוּי סוּבְּיֶקְטִיבִי: שִׁקוּל דַעַת וְהַעֲרָכַת מַצָב.

ש: עַכְשָׁיו הַלוֹגִיסְטִיקָה.

א: כַּמָה מִסְפָּרִים. בְּחֶבְרוֹת הַיְטֵק, עֲלוּיוֹת הַשִׁלוּחַ וְהַהוֹבָלָה הֵן בְּעֵרֶךְ שְׁנַיִם עַד חֲמִשָׁה אֲחוּזִים מֵהַהוֹצָאוֹת. בִּיבוּאָנִיוֹת וּבַחֲבָרוֹת תִפְעוּלִיוֹת, יוֹתֵר. וּכְשֶׁמוֹצִיאִים חֲמִשָׁה עַד עֲשָׂרָה אֲחוּזִים עַל לוֹגִיסְטִיקָה, צָרִיךְ לִבְדֹק אֵיפֹה הַכֶּסֶף, לְמָשָׁל לַעֲבֹר מֵהוֹבָלָה אֲוִירִית לְיַמִית.
א: וּכְשֶׁמַעֲבִירִים מַחְסָן לְסַפֵּק לוֹגִיסְטִי, כְּדֵי לַהֲפֹךְ הוֹצָאָה קְבוּעָה לַמִשְׁתַנֶה, מְשַׁלְמִים לוֹ לְפִי הַשֶׁטַח, הַנֶפַח, וּמִסְפַּר הַטְרַנְזְקֵצִיוֹת.

ש: וְהָיָה הֶבְדֵל בֵּין מַדָד תִפְעוּלִי לְפִינַנְסִי.

א: כֵּן. כְּשֶׁמוֹדְדִים חֹסֶר וְעֹדֶף בַּמְלַאי, הַמַדָד הַתִפְעוּלִי מוֹדֵד אֶת הַדִיוּק הַפִיזִי, בִּיחִידוֹת. הַמַדָד הַפִינַנְסִי מוֹדֵד אֶת הַשֹׁוִי הַכַּסְפִּי. חֹסֶר שֶׁל עֲשָׂרָה בְּרָגִים וְשֶׁל עֲשָׂרָה מְנוֹעִים, זֵהֶה בַּמַדָד הַתִפְעוּלִי, וְשׁוֹנֶה מְאוֹד בְּפִינַנְסִי.

ש: וָאֳנִיוֹת?

א: אַרְבָּעָה סוּגִים. אֳנִיוֹת מְכוּלוֹת, מְכָלִיוֹת לְנוֹזְלִים כְּמוֹ נֵפְט, אֳנִיוֹת צוֹבֵר לַמִטְעָן בְּתִפְזֹרֶת כְּמוֹ חִטָה, וָאֳנִיוֹת גִלְגוּל, רוֹ רוֹ, שֶׁכְּלֵי רֶכֶב וּמַשָׂאִיוֹת נוֹסְעִים לְתוֹכָן עַל הַגַלְגַלִים.
א: וְהָאֵרוּעִים בְּיַם סוּף וּבְבַאבּ אֶל מְנַדֵב הֵם דֻגְמָה לְמָה שֶׁנִקְרָא סַפְּלַיי צֵ'יין דִיסְרַפְּשֶׁן. הָאֳנִיוֹת מְשַׁנוֹת נָתִיב, הָאַסְפָּקָה מִתְעַכֶּבֶת, וְהָעֲלוּיוֹת קוֹפְצוֹת, עַד הַמְחִיר לַצַרְכָן.

ש: וּמָה עִם בִּלְעָדִיוּת?

א: לָקוֹחַ יָכוֹל לְבַקֵשׁ מֵהַסַפָּק הַלוֹגִיסְטִי שֶׁלֹא יַחֲזִיק אֶת הַסְחוֹרָה שֶׁלוֹ בְּאוֹתוֹ מִרְלוּג עִם הַמִתְחָרִים. הַסַפָּק מַעֲדִיף לְשַׁתֵף, כִּי זֶה יִתְרוֹן לְגֹדֶל. וּשְׁלֹמֹה אָמַר שֶׁגַם בְּלִי בִּלְעָדִיוּת זֶה יָכוֹל לְהִשְׁתַלֵם, כֹּל עוֹד הַסַפָּק מִתְחַיֵב לְרָמַת שֵׁרוּת גְבוֹהָה.

ש: וְהַזְמָנוֹת לָקוֹחַ?

א: שְׁנֵי כְּלָלֵי תִזְמוּן שֶׁלֹא דִבַּרְנוּ עֲלֵיהֶם. קִיט מָלֵא: לֹא מַתְחִילִים הַזְמָנָה עַד שֶׁכֹּל הֶחֳמָרִים הִגִיעוּ. הַיִתְרוֹן, שֶׁאֵין הַזְמָנוֹת תְקוּעוֹת בָּאֶמְצַע. הַחִסָרוֹן, שֶׁהַזְמָנָה שְׁלֵמָה מְחַכָּה בִּגְלַל בֹּרֶג אֶחָד שֶׁחָסֵר.
א: וְהַשֵׁנִי, שֹׁוִי הַזְמָנָה: קֹדֶם עוֹשִׂים אֶת מָה שֶׁהֲכִי רִוְחִי. הַיִתְרוֹן, הַכְנָסוֹת. הַחִסָרוֹן, שֶׁהַהַזְמָנוֹת הַקְטַנוֹת נִדְחָקוֹת שׁוּב וָשׁוּב, וְנוֹצָר זָנָב אָרֹךְ שֶׁל הַזְמָנוֹת תְקוּעוֹת.

ש: וְנִשְׁאֲרוּ הַמוֹדֵלִים שֶׁל שְׁלֹמֹה.

א: נָכוֹן, וְזֶה הַחֵלֶק הֲכִי אִישִׁי. מְשֻׁלַשׁ הַזָהָב: כְּדֵי שֶׁמְשִׂימָה תַצְלִיחַ צָרִיךְ שְׁלוֹשָׁה דְבָרִים. מִישֶׁהוּ אַחְרָאִי, מְדָדִים לְפִי סְמַארְט, וְתֹכְנִית, רוֹד מֶאפּ.
א: מוֹדֵל הַשָׁנָה הָרִאשׁוֹנָה בַּתַפְקִיד: שְׁלוֹשָׁה חוֹדָשִׁים מַקְשִׁיבִים, לוֹמְדִים אֶת הָאִרְגוּן וְאֶת הַשָׂפָה, וְרוֹשְׁמִים תוֹבָנוֹת. שְׁלוֹשָׁה חוֹדָשִׁים מְתַכְנְנִים: יִעוּדִים, מְדָדִים, תֹכְנִית עֲבוֹדָה. וְרַק אַחֲרֵי חֲצִי שָׁנָה מְיַשְׂמִים. וְאַחֲרֵי שָׁנָה, כְּבָר הוֹפְכִים לְחֵלֶק מֵהָאִרְגוּן.
א: וּמוֹדֵל הַמַעְבֹּרֶת: מַעְבֹּרֶת שֶׁחוֹזֶרֶת לְכַדוּר הָאָרֶץ צְרִיכָה לְהִכָּנֵס לָאַטְמוֹסְפֵרָה בַּזָוִית הַנְכוֹנָה. חַדָה מִדַי, נִשְׂרֶפֶת. מְתוּנָה מִדַי, נִפְלֶטֶת. וְכָכָה גַם כְּנִיסָה לַתַפְקִיד.

ש: וּמָה זֶה מוֹדֵל תִשְׁעִים עָשָׂר?

א: מוֹדֵל מִשִׁעוּר הַפְּתִיחָה. תִשְׁעִים אָחוּז מִמָה שֶׁקוֹרֶה, לֹא תָלוּי בָּנוּ. מִתְמַקְדִים בַּעֲשָׂרָה הָאֲחוּזִים שֶׁבִּשְׁלִיטָתֵנוּ.

ש: וְעוֹד מֻשָׂג אֶחָד מֵהַשִׁעוּר הָרִאשׁוֹן.

א: דִי אֶף אֶקְס. גוּף הַהַנְדָסָה מְקַבֵּל אֶת אַב הַטִיפּוּס, וְהוֹפֵךְ אוֹתוֹ לְמַשֶׁהוּ שֶׁאֶפְשָׁר לְיַצֵר. הוּא עוֹשֶׂה תַהֲלִיךְ שֶׁנִקְרָא דִיזַיְן פוֹר אֶקְסֶלֶנְס, וְאֶת הָאוֹת אֶקְס מַחֲלִיפִים בַּיַעַד שֶׁרוֹצִים לִמְקַסֵם כְּבָר בִּשְׁלַב הַתִכְנוּן, לְמָשָׁל יִצוּר אוֹ עֲלוּת.

ש: וְעַכְשָׁיו שְׁאֵלוֹת לַדֶרֶךְ. מָה הַהֶבְדֵל בֵּין חֲלָקִים קְנוּיִים לְבֵין תוֹצֶרֶת בַּתַהֲלִיךְ?
~5
א: חֲלָקִים קְנוּיִים נִקְנִים מְסַפֵּק, וּמַרְכִּיבִים מֵהֶם מִכְלוֹלִים. תוֹצֶרֶת בַּתַהֲלִיךְ נִמְצֵאת בֵּין תַחֲנוֹת הָעֲבוֹדָה בְּקַו הַיִצוּר.

ש: מָה מְבַטֵאת עֲלוּת הַחֹסֶן?
~4
א: מָה יִקְרֶה, וּמָה זֶה יַעֲלֶה, אִם יֶחְסַר פְּרִיט.

ש: לְאֵיזוֹ קְבוּצָה מַתְאִימָה הַמְגֵרָה הַכְּפוּלָה?
~4
א: לִפְרִיטֵי סִי, הַזוֹלִים וְהַפָּחוֹת חֲשׁוּבִים.

ש: יְבוּאָן מֵבִיא מַשָׂאִיוֹת שֶׁנוֹסְעוֹת לְתוֹךְ הָאֳנִיָה. אֵיזוֹ אֳנִיָה?
~4
א: אֳנִיַת גִלְגוּל, רוֹ רוֹ.

ש: וּמָה הַחִסָרוֹן שֶׁל קִיט מָלֵא?
~4
א: הַזְמָנָה שְׁלֵמָה מְחַכָּה בִּגְלַל רְכִיב קָטָן אֶחָד שֶׁחָסֵר.

ש: כֹּל הַשְׁאֵלוֹת הָאֵלֶה נִמְצָאוֹת עַכְשָׁיו בַּאֲתַר הַתִרְגוּל, בַּנוֹשְׂאִים הָרְגִילִים, וּבַסוֹף כֹּל הֶסְבֵּר כָּתוּב מֵאֵיזֶה שִׁעוּר הֵן. בְּהַצְלָחָה בַּמִבְחָן.

א: בְּהַצְלָחָה.
