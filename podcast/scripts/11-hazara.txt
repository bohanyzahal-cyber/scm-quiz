# title: חזרה למבחן — מלכודות החישוב וטעויות בשקפים
# פרק סיכום. כל עובדה כאן מופיעה בפרקים 1–10, בבנקים, בדף הנוסחאות, ב-README (טעויות בשקפים) או בתמלולי ההקלטות.
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק אַחֲרוֹן: חֲזָרָה לְמִבְחָן. מָה אֲנַחְנוּ יוֹדְעִים עַל הַמִבְחָן עַצְמוֹ?

א: עֶשְׂרִים וְחָמֵשׁ שְׁאֵלוֹת אָמֵרִיקָאִיוֹת, כַּמוּתִיוֹת וְאֵיכוּתִיוֹת, עַל הַחֹמֶר שֶׁל שְׁנֵי הַמַרְצִים, וְהַמִבְחָן שָׁוֶה שִׁשִׁים אָחוּז מֵהַצִיוּן. נִצָן אָמַר שֶׁיִהְיוּ חִשׁוּבִים, אֲבָל לֹא מְסֻבָּכִים. אָז מְבִיאִים מַחְשְׁבוֹן.
א: וּשְׁתֵי הִסְתַיְגוּיוֹת. בַּאֲתַר הַקוּרְס אֵין מִקוּד וְאֵין שְׁאֵלוֹת לְדֻגְמָה. וְהַהַקְלָטוֹת מֵהַשְׁמוֹנָה עָשָׂר בְּאוֹגוּסְט וָאֵילָךְ לֹא זְמִינוֹת לַסְטוּדֶנְטִים, כָּךְ שֶׁיִתָכֵן שֶׁהַמַרְצִים הִדְגִישׁוּ שָׁם דְבָרִים שֶׁלֹא הִגִיעוּ אֵלֵינוּ. וְלָכֵן כְּדַאי לְתַרְגֵל אֶת כֹּל הַחֹמֶר.

ש: מָה כֵּן שָׁמַעְנוּ בְּהַקְלָטוֹת עַל הֶקֵף הַחֹמֶר?

א: בְּחִזוּי, מָה שֶׁנִלְמַד בְּפֹעַל: מְמֻצָע נָע, מְמֻצָע נָע מְשֻׁקְלָל, וְרֶגְרֶסְיָה, עִם שְׁנֵי מַדָדֵי טָעוּת בִּלְבַד: הַסְטִיָה הַמֻחְלֶטֶת הַמְמֻצַעַת, וּמְמֻצַע רִבּוּעֵי הַסְטִיוֹת. מַדַד הָאֲחוּזִים שֶׁהוֹפִיעַ בְּמַצֶגֶת? הַמַרְצֶה אָמַר לְהִתְעַלֵם מִמֶנוּ.
א: בְּחִזוּי עוֹנָתִי, נִצָן אָמַר בִּמְפוֹרָשׁ שֶׁהַדֵקוֹמְפּוֹזִיצְיָה וְהַשִׁיטָה הַפְּשׁוּטָה לֹא יִהְיוּ בְּמִבְחַן. מָה שֶׁעָשׂוּי לְהוֹפִיעַ הוּא הַשְׁאֵלָה אִם יֵשׁ עוֹנָתִיוֹת, לְפִי כְּלַל הָאֶצְבַּע: לְפָחוֹת מְקַדֵם מְמֻצָע אֶחָד מֵעַל אַחַת נְקֻדָה שְׁתַיִם, אוֹ מִתַחַת לְאֶפֶס נְקֻדָה שְׁמוֹנָה.
א: וְעוֹד שְׁלוֹשָׁה עֶקְרוֹנוֹת מֵהַהַרְצָאוֹת. שָׁנָה חֲרִיגָה, כְּמוֹ יְרִידָה שֶׁל אַרְבָּעִים אָחוּז בִּגְלַל מִלְחָמָה, מְנַרְמְלִים. טָעוּת הַקְלָדָה מוֹרִידִים. וּכְשֶׁיֵשׁ מְגַמָה, לוֹקְחִים מְעַט תְקוּפוֹת, וְעָדִיף רֶגְרֶסְיָה. הַמְמֻצָע הַנָע מַתְאִים לְבִקוּשׁ קָבוּעַ.

ש: עַכְשָׁו רְשִׁימַת בְּדִיקָה לְחִשׁוּבִים. נַתְחִיל מֵהַחִזוּי.

א: מְמֻצָע נָע עַל חַלוֹן שֶׁל שָׁלוֹשׁ תְקוּפוֹת: לִשְׁלוֹשׁ הַתְקוּפוֹת הָרִאשׁוֹנוֹת אֵין תַחֲזִית. וְלָכֵן גַם מַדָדֵי הַטָעוּת מְחֻלָקִים רַק בְּמִסְפַּר הַתְקוּפוֹת שֶׁיֵשׁ לָהֶן תַחֲזִית.
א: בִּמְמֻצָע מְשֻׁקְלָל, לְפִי נִצָן: לַתְקוּפָה הָאַחֲרוֹנָה מִשְׁקָל אַחַת, וּבְכֹל תְקוּפָה אָחוֹרָה מַפְחִיתִים עֲשִׂירִית. וּמְחַלְקִים בִּסְכוּם הַמִשְׁקָלוֹת שֶׁל הַחַלוֹן, שֶׁלֹא חַיָב לִהְיוֹת אַחַת.
א: בְּרֶגְרֶסְיָה שֶׁל סִדְרַת זְמַן, צִיר אִיקְס הוּא מִסְפַּר הַתְקוּפָה, וְלֹא הַשָׁנָה. וּמְקַדֵם הַמְתָאֵם הוּא אָר, וְלֹא אָר בָּרִבּוּעַ.
א: וּמְקַדְמֵי עוֹנָתִיוֹת: הַסְכוּם שֶׁלָהֶם בַּמַחְזוֹר שָׁוֶה לְמִסְפַּר הָעוֹנוֹת. בְּאַרְבָּעָה רִבְעוֹנִים, אַרְבַּע.

ש: נִהוּל מְלַאי.

א: בַּכַּמוּת הַכַּלְכָּלִית לַהַזְמָנָה: עֲלוּת הָאַחְזָקָה שֶׁל יְחִידָה הִיא מְחִיר הַפְּרִיט כָּפוּל מְקַדֵם הָאַחְזָקָה. הַצְרִיכָה, שְׁנָתִית. וּבַנְקֻדָה הָאוֹפְּטִימָלִית, כְּשֶׁאֵין אִלוּצִים, עֲלוּת הָאַחְזָקָה וַעֲלוּת הַהַזְמָנוֹת שָׁווֹת. אִם לֹא יָצְאוּ שָׁווֹת, כְּדַאי לִבְדֹק אֶת הַחִשׁוּב.
א: וּבִמְלַאי בִּטָחוֹן, שָׁלוֹשׁ מַלְכּוֹדוֹת. הָרִאשׁוֹנָה: זְמַן הָאַסְפָּקָה בְּתוֹךְ הַשֹׁרֶשׁ, בְּאוֹתָה יְחִידַת זְמַן שֶׁל סְטִיַת הַתֶקֶן. בַּחֹדֶשׁ יֵשׁ אַרְבָּעָה נְקֻדָה שְׁלוֹשָׁה שָׁבוּעוֹת, אוֹ שְׁלוֹשִׁים יָמִים. הַשְׁנִיָה: סְטִיַת תֶקֶן מִדְגָמִית. וְהַשְׁלִישִׁית: לָקַחַת אֶת קָיי מֵהַטַבְלָה שֶׁל נִצָן. לְתִשְׁעִים וַחֲמִשָׁה אָחוּז, אַחַת נְקֻדָה תִשְׁעִים וָשֵׁשׁ. לְתִשְׁעִים, אַחַת נְקֻדָה שִׁשִׁים וְחָמֵשׁ.

ש: וּנְקֻדַת הַהַזְמָנָה?

א: שְׁתֵי נֻסְחָאוֹת, לְפִי הַיַחַס בֵּין זְמַן הָאַסְפָּקָה לִזְמַן הַמַחְזוֹר. כְּשֶׁהָאַסְפָּקָה קְצָרָה מֵהַמַחְזוֹר: מְלַאי הַבִּטָחוֹן, וְעוֹד הַכַּמוּת כָּפוּל זְמַן הָאַסְפָּקָה חֶלְקֵי זְמַן הַמַחְזוֹר. כְּשֶׁהִיא אֲרֻכָּה: מְלַאי הַבִּטָחוֹן, וְעוֹד הַצְרִיכָה בִּזְמַן הָאַסְפָּקָה, פָּחוֹת הַכַּמוּת כָּפוּל מִסְפַּר הַמַחְזוֹרִים הַשְׁלֵמִים, מְעֻגָל כְּלַפֵּי מַטָה.
א: וְהַהִסְתַבְּרוּת לְחֹסֶר: הַמַשְׁלִים שֶׁל רָמַת הַזְמִינוּת, חֶלְקֵי שְׁתַיִם. כְּשֶׁהַזְמִינוּת הִיא תִשְׁעִים אָחוּז, הַהִסְתַבְּרוּת לְחֹסֶר הִיא חֲמִשָׁה אֲחוּזִים.

ש: פָּארֶטוֹ וְאַחְזָקָה.

א: בְּפָּארֶטוֹ, מְדָרְגִים לְפִי עֵרֶךְ הַצְרִיכָה הַשְׁנָתִית, כְּלוֹמַר צְרִיכָה כָּפוּל מְחִיר, וְלֹא לְפִי הַכַּמוּת. וּשְׁמוֹנִים וְעֶשְׂרִים לֹא חַיָבִים לְהַשְׁלִים לְמֵאָה. וּבְפִלוּחַ רַב מְמַדִי, מְנַרְמְלִים: תְלוּנוֹת חֶלְקֵי מְבַקְרִים.
א: בַּאֲמִינוּת: בְּטוּר מַכְפִּילִים. בְּמַקְבִּיל, אַחַת פָּחוֹת מַכְפֵּלַת הַהִסְתַבְּרוּיוֹת לְכֶשֶׁל. וְהַמַעֲרֶכֶת בְּטוּר לֹא אֲמִינָה יוֹתֵר מֵהָרְכִיב הַחַלָשׁ שֶׁלָה.
א: בִּתְדִירוּת הָאַחְזָקָה הַמוֹנַעַת: אִם הַתִקוּן מִתְקַצֵר, כְּדַאי פָּחוֹת טִפּוּלִים. אִם הַטִפּוּל מִתְקַצֵר, כְּדַאי יוֹתֵר.
א: וּבִנְקֻדַת הָאִזוּן בֵּין שְׁתֵי חֲלוּפוֹת: מַשְׁוִים אֶת כַּמוּת שִׁיווּי הַמִשְׁקָל לַכַּמוּת שֶׁל כֹּל הַפְּרוֹיֵקְט, וְלֹא לְכַמוּת שֶׁל שָׁנָה אַחַת.

ש: וְעַכְשָׁו הַטָעוּיוֹת בַּשְׁקָפִים. לָמָה זֶה חָשׁוּב?

א: כִּי מִי שֶׁמַחְשֵׁב נָכוֹן וּמְקַבֵּל מִסְפָּר אַחֵר מֵהַשֶׁקֶף, עָלוּל לַחֲשֹׁב שֶׁהוּא טָעָה. אָז הִנֵה הָרְשִׁימָה.
א: בַּחִזוּי, בַּשֶׁקֶף הַשְׁוָאַת הַשִׁיטוֹת, שׁוּרַת הַמְמֻצָע הַנָע הַמְשֻׁקְלָל נִלְקְחָה מִסִדְרַת נְתוּנִים אַחֶרֶת. הַמַסְקָנָה, שֶׁהָרֶגְרֶסְיָה עֲדִיפָה, לֹא מִשְׁתַנָה.
א: בְּמַצֶגֶת הַמְלַאי, בַּשֶׁקֶף תֵשַׁע, שְׁנֵי הַשְׁבָרִים שֶׁל סְבָבֵי הַמְלַאי וְהַמְלַאי בַּשָׁבוּעוֹת כְּתוּבִים הָפוּךְ.
א: בַּמְלַאי בִּטָחוֹן, בַּשֶׁקֶף עָשָׂר, הַתוֹצָאָה חֻשְׁבָה בְּלִי הַשֹׁרֶשׁ: עֶשְׂרִים וְאַרְבַּע בִּמְקוֹם שְׁלוֹשִׁים וּשְׁתַיִם. וּבְשֶׁקֶף חֲמֵשׁ עֶשְׂרֵה, סְטִיַת הַתֶקֶן הִיא שְׁלוֹשׁ מֵאוֹת שִׁשִׁים וְחָמֵשׁ נְקֻדָה שֵׁשׁ, וְלֹא שְׁלוֹשׁ מֵאוֹת חֲמִשִׁים וְשֵׁשׁ נְקֻדָה שֵׁשׁ.
א: בְּפָּארֶטוֹ, בַּשֶׁקֶף שֵׁשׁ עֶשְׂרֵה, שְׁמוֹת הַסְנִיפִים בְּטַבְלַת הָעֹמֶס הִתְחַלְפוּ.
א: בְּמַצֶגֶת הָאַחְזָקָה, בַּשֶׁקֶף שְׁלוֹשִׁים וְחָמֵשׁ, צָרִיךְ לִהְיוֹת אַרְבָּעִים וְאַרְבָּעָה אֶלֶף וְאַרְבַּע מֵאוֹת, וְלֹא אַרְבָּעָה עָשָׂר אֶלֶף וְאַרְבַּע מֵאוֹת.
א: בָּאֲמִינוּת, הַחִשׁוּבִים שֶׁל הַמַקְבִּיל בִּשְׁקָפִים תְשַׁע עֶשְׂרֵה וְעֶשְׂרִים, וְשֶׁל הַגִבּוּי בַּשֶׁקֶף עֶשְׂרִים וְשָׁלוֹשׁ, לֹא מְדֻיָקִים. עִם שְׁנֵי הַמְנוֹעִים יוֹצֵא תִשְׁעִים וְתִשְׁעָה נְקֻדָה חֲמִשִׁים וַחֲמִשָׁה אָחוּז.
א: וּבִתְדִירוּת הָאַחְזָקָה הַמוֹנַעַת: בַּשֶׁקֶף עָשָׂר, הַלוֹגָרִיתְם שֶׁל שְׁלוֹשִׁים הוּא שָׁלוֹשׁ נְקֻדָה אַרְבָּעִים, וְלֹא אַרְבַּע נְקֻדָה שֵׁשׁ. וּבְשֶׁקֶף שְׁתֵים עֶשְׂרֵה, בַּנֻסְחָה כָּתוּב שָׁלוֹשׁ בִּמְקוֹם אַרְבַּע, אֲבָל הַתוֹצָאָה נְכוֹנָה.

ש: וּמָה עוֹשִׂים עִם זֶה בְּמִבְחַן?

א: מְחַשְׁבִים לְפִי הַנֻסְחָה. בְּדַף הַנֻסְחָאוֹת בַּאֲתַר כֹּל הַדֻגְמָאוֹת הָאֵלֶה פְּתוּרוֹת, עִם הַחִשׁוּב הַנָכוֹן.

ש: עוֹבְרִים לִשְׁלֹמֹה. אֵילוּ מֻשָׂגִים הֲכִי קַל לְבַלְבֵּל?

א: בּוֹאִי נַעֲשֶׂה סֶבֶב מָהִיר. אַתְ שׁוֹאֶלֶת, וַאֲנִי עוֹנָה בִּקְצָרָה.

ש: מוּכָנָה. מָתַי נִגְמַר תַהֲלִיךְ הָאַסְפָּקָה?

א: כְּשֶׁהַלָקוֹחַ מִשְׁתַמֵשׁ בַּמוּצָר. לֹא כְּשֶׁהַמִשְׁלוֹחַ יוֹצֵא.

ש: שְׁבִירַת הַתַהֲלִיךְ, מֻשָׂג חִיוּבִי אוֹ שְׁלִילִי?

א: חִיוּבִי. חֲלֻקָה לִתְחוּמֵי אַחְרָיוּת בְּרוּרִים.

ש: מְלַאי נָמוּךְ נִמְצָא בַּיִעוּד אוֹ בָּאֶתְגָר?

א: בָּאֶתְגָר. וּבְאַרְגַז הַכֵּלִים נוֹסְפָה לָאֶתְגָר גַם אֵיכוּת גְבוֹהָה.

ש: מָה חָסֵר בַּחֲמֵשֶׁת הַנְכוֹנִים שֶׁל הָרֶכֶשׁ?

א: הַמְחִיר.

ש: אֲחוּז חִסָכוֹן מְחַלְקִים בַּמֶה?

א: בִּמְחִיר הַבָּסִיס. לְמוּצָר חָדָשׁ, הַצָעַת הַמְחִיר הַכְּתוּבָה הָרִאשׁוֹנָה.

ש: סֵדֶר הַפִּירָמִידָה שֶׁל מִקוּר הַחוּץ?

א: הַכֹּל בַּבַּיִת, קַבְּלָנוּת מִשְׁנֶה, מִקוּר חוּץ, מִקוּר חוּץ מָלֵא, וְאָז שָׁלוֹשׁ רָמוֹת עִם תִכְנוּן. וּבַפִּסְגָה, הַסַפָּק שֶׁמְתַכְנֵן וּמְיַצֵר, וְהַחֶבְרָה רַק שָׂמָה מוּתָג.

ש: וּבְלוֹגִיסְטִיקָה?

א: טוּ פִּי אֶל הִיא קַבְּלָנוּת מִשְׁנֶה. תְרִי פִּי אֶל הִיא מִקוּר חוּץ. פוֹר פִּי אֶל הִיא מִקוּר חוּץ מָלֵא.

ש: שְׁלֹשֶׁת הַמֻשָׂגִים הַיַפָּנִיִים?

א: טַאקְט טַיים, אִזוּן קַוֵי הַיִצוּר. הַיי ג'וּנְקָה, אִזוּן הָאַסְפָּקוֹת מוּל הַבִּקוּשׁ. ג'ִידוֹקָה, אוֹטוֹמַצְיָה חֲכָמָה.

ש: תַעֲבוּרָה אוֹ תְנוּעוֹת מְיֻתָרוֹת?

א: תַעֲבוּרָה הִיא שִׁנוּעַ שֶׁל חוֹמָרִים. תְנוּעוֹת מְיֻתָרוֹת הֵן שֶׁל הָעוֹבֵד עַצְמוֹ.

ש: מַטְרִיצַת הַגְמִישׁוּת?

א: הַגְדָלָה: עֶשְׂרֵה, עֶשְׂרִים, שְׁלוֹשִׁים. הַקְטַנָה: חֲמִשָׁה עָשָׂר, עֶשְׂרִים וַחֲמִשָׁה, שְׁלוֹשִׁים וַחֲמִשָׁה. וּבַחֹדֶשׁ הַנוֹכְחִי, אֵין שִׁנוּי.

ש: אֵיי בִּי סִי אֵצֶל שְׁלֹמֹה, וְאֵצֶל נִצָן?

א: בִּשְׁנֵיהֶם, קְבוּצָה אֵיי הִיא כִּשְׁמוֹנִים אָחוּז מֵהָעֵרֶךְ. אֲבָל אֵצֶל שְׁלֹמֹה הִיא כַּעֲשָׂרָה אֲחוּזִים מֵהַכַּמוּת, וְאֵצֶל נִצָן כַּחֲמִשָׁה אֲחוּזִים מֵהַפְּרִיטִים.

ש: יְמֵי אִחוּר מְמֻצָעִים מְחַלְקִים בַּמֶה?

א: בְּמִסְפַּר הַהַזְמָנוֹת הַמְאַחֲרוֹת בִּלְבַד.

ש: וַעֲמִידָה בַּזְמַנִים, עַל הַזְמָנָה אוֹ עַל שׁוּרָה?

א: שְׁתֵי הָאֶפְשָׁרוּיוֹת. הַזְמָנָה עִם שׁוּרָה אַחַת מְאַחֶרֶת הִיא אֶפֶס אָחוּז לְפִי הַהַזְמָנָה, אֲבָל לְפִי הַשׁוּרוֹת הִיא יְכוֹלָה לִהְיוֹת שְׁמוֹנִים.

ש: בִּקוּשׁ לֹא צָפוּי וּזְמַן אַסְפָּקָה אָרֹךְ?

א: לְגַדֵר בְּאֶמְצָעוּת מְלַאי.

ש: וּבְאֵיזֶה רִבְעוֹן בַּשָׁנָה הָרִאשׁוֹנָה מַחֲלִיפִים כּוֹחַ אָדָם?

א: בַּשְׁלִישִׁי. בָּרִאשׁוֹן מַקְשִׁיבִים, בִּשְׁנֵי מְתַכְנְנִים, בָּרְבִיעִי מְבַקְרִים וְלוֹמְדִים.

ש: יָפֶה. וּמָה הֲכִי חָשׁוּב לִזְכֹּר בַּסוֹף?

א: שְׁלֹמֹה חוֹזֵר עַל מִשְׁפָּט אֶחָד בְּכַמָה מַצָגוֹת: הַגְדָלַת הַמְלַאי אֵינָה הַפִּתְרוֹן. וְנִצָן אָמַר: עַל רֹב הַשְׁאֵלוֹת, הַתְשׁוּבָה הִיא זֶה תָלוּי. כְּלוֹמַר, בִּשְׁאֵלָה אֵיכוּתִית, מְחַפְּשִׂים אֶת הַתְשׁוּבָה שֶׁמַתְאִימָה לַנְסִבּוֹת שֶׁבַּתַרְחִישׁ.

ש: וְאֵיךְ מִתְרַגְלִים?

א: בַּאֲתַר הַתִרְגוּל יֵשׁ יוֹתֵר מֵחָמֵשׁ מֵאוֹת שְׁאֵלוֹת, מְסֻדָרוֹת לְפִי נוֹשְׂאִים, עִם הֶסְבֵּר לְכֹל תְשׁוּבָה. יֵשׁ סִימוּלַצְיָה שֶׁל מִבְחָן, עֶשְׂרִים וְחָמֵשׁ שְׁאֵלוֹת כְּמוֹ בַּמִבְחָן הָאֲמִתִי. וְיֵשׁ מַצָב שֶׁל חִשׁוּבִים בִּלְבַד, וְדַף נֻסְחָאוֹת עִם כֹּל הַדֻגְמָאוֹת פְּתוּרוֹת.

ש: אָז זֶהוּ. זֶה הָיָה הַפֶּרֶק הָאַחֲרוֹן. בְּהַצְלָחָה בַּמִבְחָן.

א: בְּהַצְלָחָה.
