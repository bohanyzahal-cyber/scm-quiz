# title: תדירות אחזקה מונעת ונקודת איזון
# מצגות 9, 10 ו-11 של ניצן. כל עובדה כאן מופיעה בבנקים 19 ו-20 או בדף הנוסחאות.
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.
# הסמלים היווניים (למבדה, מיו, פי) מתוארים במילים: קצב התקלות, קצב התיקונים האפשרי, קצב הטיפולים האפשרי.

ש: פֶּרֶק שִׁשִׁי: שְׁנֵי מוֹדֵלִים שֶׁבְּנוּיִים עַל אוֹתוֹ רַעְיוֹן, אִזוּן. כַּמָה טִפּוּלִים מוֹנְעִים לַעֲשׂוֹת בְּחֹדֶשׁ, וּמָתַי מוּצָר אוֹ הַשְׁקָעָה מַתְחִילִים לְהַרְוִיחַ.

א: נַתְחִיל מֵהַטִפּוּלִים הַמוֹנְעִים. גַם תִקוּנֵי שֶׁבֶר וְגַם טִפּוּלִים מוֹנְעִים מַשְׁבִּיתִים אֶת הַיִצוּר. מְעַט מִדַי טִפּוּלִים מוֹנְעִים, וְהַתַקָלוֹת מִתְרַבּוֹת. יוֹתֵר מִדַי טִפּוּלִים, וְהַטִפּוּלִים עַצְמָם יוֹצְרִים הַשְׁבָּתָה. בָּאֶמְצַע יֵשׁ מִסְפָּר אוֹפְּטִימָלִי.

ש: מָה הַמִשְׁתַנִים?

א: מִסְפַּר הַטִפּוּלִים הַמוֹנְעִים בְּחֹדֶשׁ הוּא מִשְׁתַנֶה הַהַחְלָטָה. קֶצֶב הַתַקָלוֹת תָלוּי בּוֹ: כְּכֹל שֶׁעוֹשִׂים יוֹתֵר טִפּוּלִים, יֵשׁ פָּחוֹת תַקָלוֹת.
א: וְיֵשׁ שְׁנֵי קְצָבִים אֶפְשָׁרִיִים. קֶצֶב הַתִקוּנִים הָאֶפְשָׁרִי: כַּמָה תַקָלוֹת אֶפְשָׁר לְתַקֵן בִּיחִידַת זְמַן, כְּלוֹמַר אַחַת חֶלְקֵי מֶשֶׁךְ תִקוּן. וְקֶצֶב הַטִפּוּלִים הָאֶפְשָׁרִי: כַּמָה טִפּוּלִים מוֹנְעִים אֶפְשָׁר לְבַצֵעַ בִּיחִידַת זְמַן, כְּלוֹמַר אַחַת חֶלְקֵי מֶשֶׁךְ טִפּוּל.

ש: וְאֵיךְ נִרְאֶה זְמַן הַהַשְׁבָּתָה?

א: שְׁנֵי חֲלָקִים. הַשְׁבָּתָה בִּגְלַל תַקָלוֹת: מִסְפַּר הַתַקָלוֹת, כָּפוּל מֶשֶׁךְ תִקוּן. וְהַשְׁבָּתָה בִּגְלַל טִפּוּלִים: מִסְפַּר הַטִפּוּלִים, כָּפוּל מֶשֶׁךְ טִפּוּל. וּזְמַן הַהַשְׁבָּתָה הַכּוֹלֵל הוּא הַסְכוּם.

ש: וְאֵיךְ מוֹצְאִים אֶת הַמִינִימוּם?

א: גוֹזְרִים לְפִי מִסְפַּר הַטִפּוּלִים וּמַשְׁוִים לְאֶפֶס. וּבַמִקְרֶה הַפְּרָטִי שֶׁבַּמַצֶגֶת, מִסְפַּר הַתַקָלוֹת הוּא אֶף כָּפוּל אִי בְּחֶזְקַת מִינוּס מִסְפַּר הַטִפּוּלִים. אֶף הוּא מִסְפַּר הַתַקָלוֹת כְּשֶׁלֹא עוֹשִׂים טִפּוּלִים מוֹנְעִים בִּכְלָל.
א: וְאָז הַפִּתְרוֹן יוֹצֵא פָּשׁוּט: מִסְפַּר הַטִפּוּלִים הָאוֹפְּטִימָלִי שָׁוֶה לַלוֹגָרִיתְם הַטִבְעִי שֶׁל: אֶף, כָּפוּל קֶצֶב הַטִפּוּלִים הָאֶפְשָׁרִי, חֶלְקֵי קֶצֶב הַתִקוּנִים הָאֶפְשָׁרִי.

ש: בּוֹא נַעֲבֹר לְדֻגְמָה.

א: תִקוּן תַקָלָה לוֹקֵחַ שְׁתֵים עֶשְׂרֵה שָׁעוֹת, טִפּוּל מוֹנֵעַ שָׁלוֹשׁ שָׁעוֹת, וְיֵשׁ מָאתַיִם שְׁעוֹת עֲבוֹדָה בְּחֹדֶשׁ. וּבַחֹדֶשׁ שֶׁבּוֹ לֹא עָשׂוּ טִפּוּלִים מוֹנְעִים, הָיוּ אַרְבַּע תַקָלוֹת.
א: קֶצֶב הַטִפּוּלִים הָאֶפְשָׁרִי: מָאתַיִם חֶלְקֵי שָׁלוֹשׁ, שִׁשִׁים וָשֵׁשׁ נְקֻדָה שִׁשִׁים וְשֶׁבַע בְּחֹדֶשׁ. קֶצֶב הַתִקוּנִים הָאֶפְשָׁרִי: מָאתַיִם חֶלְקֵי שְׁתֵים עֶשְׂרֵה, שֵׁשׁ עֶשְׂרֵה נְקֻדָה שִׁשִׁים וְשֶׁבַע.
א: אַרְבַּע כָּפוּל שִׁשִׁים וְשֵׁשׁ נְקֻדָה שִׁשִׁים וְשֶׁבַע, חֶלְקֵי שֵׁשׁ עֶשְׂרֵה נְקֻדָה שִׁשִׁים וְשֶׁבַע, זֶה שֵׁשׁ עֶשְׂרֵה. וְהַלוֹגָרִיתְם הַטִבְעִי שֶׁל שֵׁשׁ עֶשְׂרֵה: שְׁתַיִם נְקֻדָה שִׁבְעִים וְשֶׁבַע טִפּוּלִים בְּחֹדֶשׁ.

ש: וְכַמָה שְׁעוֹת הַשְׁבָּתָה יוֹצְאוֹת?

א: עִם שְׁתַיִם נְקֻדָה שִׁבְעִים וְשֶׁבַע טִפּוּלִים, נִשְׁאֶרֶת רֶבַע תַקָלָה בְּחֹדֶשׁ בִּמְמֻצָע. רֶבַע כָּפוּל שְׁתֵים עֶשְׂרֵה שָׁעוֹת, זֶה שָׁלוֹשׁ שָׁעוֹת. וְעוֹד שְׁתַיִם נְקֻדָה שִׁבְעִים וְשֶׁבַע כָּפוּל שָׁלוֹשׁ שָׁעוֹת, שְׁמוֹנָה נְקֻדָה שְׁלוֹשִׁים וּשְׁתַיִם. בְּסַךְ הַכֹּל, אַחַת עֶשְׂרֵה שָׁעוֹת וּשְׁלִישׁ בְּחֹדֶשׁ. וּבְלִי טִפּוּלִים מוֹנְעִים? אַרְבַּע תַקָלוֹת כָּפוּל שְׁתֵים עֶשְׂרֵה, אַרְבָּעִים וּשְׁמוֹנֶה שָׁעוֹת.

ש: וְיֵשׁ בַּשְׁקָפִים שְׁתֵי טָעוּיוֹת.

א: כֵּן. בַּשֶׁקֶף עָשָׂר, עִם אֶף שָׁלוֹשׁ, קֶצֶב טִפּוּלִים שֶׁל שְׁלוֹשׁ מֵאוֹת, וְקֶצֶב תִקוּנִים שֶׁל שְׁלוֹשִׁים, כָּתוּב שֶׁהַתוֹצָאָה הִיא אַרְבַּע נְקֻדָה שֵׁשׁ. אֲבָל שָׁלוֹשׁ כָּפוּל שְׁלוֹשׁ מֵאוֹת, חֶלְקֵי שְׁלוֹשִׁים, זֶה שְׁלוֹשִׁים. וְהַלוֹגָרִיתְם שֶׁל שְׁלוֹשִׁים הוּא שָׁלוֹשׁ נְקֻדָה אַרְבָּעִים. אַרְבַּע נְקֻדָה שֵׁשׁ זֶה הַלוֹגָרִיתְם שֶׁל מֵאָה.
א: וּבְשֶׁקֶף שְׁתֵים עֶשְׂרֵה, בַּנֻסְחָה כָּתוּב שָׁלוֹשׁ בִּמְקוֹם אַרְבַּע, אֲבָל הַתוֹצָאָה, שְׁתַיִם נְקֻדָה שִׁבְעִים וְשֶׁבַע, נְכוֹנָה, כִּי הִיא חֻשְׁבָה עִם אַרְבַּע.

ש: וּמָה קוֹרֶה כְּשֶׁמְשַׁנִים פָּרָמֶטְרִים?

א: הַמַצֶגֶת מְבַקֶשֶׁת נִתוּחַ רְגִישׁוּת, עִם הֶסְבֵּר לוֹגִיסְטִי. אִם יֵשׁ יוֹתֵר תַקָלוֹת בְּלִי טִפּוּלִים, כְּדַאי יוֹתֵר טִפּוּלִים. אִם טִפּוּל מוֹנֵעַ מִתְקַצֵר, הוּא זוֹל יוֹתֵר בִּזְמַן, וּכְדַאי יוֹתֵר טִפּוּלִים. וְאִם תִקוּן תַקָלָה מִתְקַצֵר, כֹּל תַקָלָה מַשְׁבִּיתָה פָּחוֹת, וְלָכֵן כְּדַאי פָּחוֹת טִפּוּלִים.

ש: וְהַמַצֶגֶת הַבָּאָה מַחְלִיפָה אֶת הַמַטָרָה: מַקְסִימוּם רֶוַח.

א: נָכוֹן. נְקֻדַת הַמוֹצָא הִיא הָרֶוַח שֶׁהָאִרְגוּן הָיָה מַשִׂיג בְּלִי הַשְׁבָּתָה בִּכְלָל, בִּתְפוּקָה מְלֵאָה. מִמֶנוּ מַפְחִיתִים אַרְבָּעָה דְבָרִים: הֶפְסֵד יִצוּר בִּזְמַן הַטִפּוּלִים, הֶפְסֵד יִצוּר בִּזְמַן הַתִקוּנִים, עֲלוּת הַטִפּוּלִים, וַעֲלוּת הַתִקוּנִים.
א: וְהַפִּתְרוֹן דוֹמֶה מְאוֹד: הַלוֹגָרִיתְם הַטִבְעִי שֶׁל אֶף, כָּפוּל קֶצֶב הַטִפּוּלִים, כָּפוּל הָרֶוַח וְעוֹד עֲלוּת הַתִקוּן, חֶלְקֵי קֶצֶב הַתִקוּנִים, כָּפוּל הָרֶוַח וְעוֹד עֲלוּת הַטִפּוּל.

ש: וְהַדֻגְמָה?

א: רֶוַח שֶׁל עֲשֶׂרֶת אֲלָפִים דוֹלָר בְּחֹדֶשׁ, אֶף שְׁתַיִם, קֶצֶב טִפּוּלִים אֶפְשָׁרִי שֶׁל מֵאָה, קֶצֶב תִקוּנִים אֶפְשָׁרִי שֶׁל עֶשְׂרִים וְחָמֵשׁ, עֲלוּת טִפּוּל שִׁבְעִים וַחֲמִשָׁה דוֹלָר, וַעֲלוּת תִקוּן אַרְבַּע מֵאוֹת. יוֹצֵא לוֹגָרִיתְם שֶׁל שְׁמוֹנָה נְקֻדָה עֶשְׂרִים וְחָמֵשׁ, כְּלוֹמַר שְׁתַיִם נְקֻדָה אַחַת עֶשְׂרֵה טִפּוּלִים בְּחֹדֶשׁ.
א: וְהָרֶוַח בַּנְקֻדָה הַזֹאת כְּתִשְׁעַת אֲלָפִים שֵׁשׁ מֵאוֹת שְׁמוֹנִים וָשֵׁשׁ דוֹלָר. לְשֵׁם הַשְׁוָאָה, בְּלִי טִפּוּלִים מוֹנְעִים בִּכְלָל, תִשְׁעַת אֲלָפִים מֵאָה שִׁשִׁים וּשְׁמוֹנֶה. כְּלוֹמַר הָאַחְזָקָה הַמוֹנַעַת מוֹסִיפָה יוֹתֵר מֵחָמֵשׁ מֵאוֹת דוֹלָר בְּחֹדֶשׁ.

ש: וּמָה מַשְׁפִּיעַ כָּאן?

א: כְּשֶׁעֲלוּת תִקוּן גְבוֹהָה, כְּדַאי יוֹתֵר טִפּוּלִים מוֹנְעִים. וּכְשֶׁעֲלוּת טִפּוּל גְבוֹהָה, כְּדַאי פָּחוֹת.

ש: עוֹבְרִים לַנוֹשֵׂא הָאַחֲרוֹן שֶׁל נִצָן: שִׁיווּי מִשְׁקָל כַּלְכָּלִי.

א: שְׁנֵי שִׁמוּשִׁים. נְקֻדַת אִזוּן בֵּין הַכְנָסוֹת לְהוֹצָאוֹת שֶׁל פְּרִיט בּוֹדֵד. וְשִׁיווּי מִשְׁקָל בֵּין חֲלוּפוֹת הַשְׁקָעָה, כְּלוֹמַר נִתוּחַ כְּדָאִיוּת.

ש: נְקֻדַת הָאִזוּן.

א: הַהַכְנָסוֹת הֵן הַמְחִיר כָּפוּל הַכַּמוּת. הַהוֹצָאוֹת הֵן עֲלוּת קְבוּעָה, וְעוֹד עֲלוּת מִשְׁתַנָה לִיחִידָה כָּפוּל הַכַּמוּת. בִּנְקֻדַת הָאִזוּן הֵן שָׁווֹת, וְלָכֵן הַכַּמוּת הִיא הָעֲלוּת הַקְבוּעָה, חֶלְקֵי הַמְחִיר פָּחוֹת הָעֲלוּת הַמִשְׁתַנָה.
א: הַמְחִיר פָּחוֹת הָעֲלוּת הַמִשְׁתַנָה הוּא מָה שֶׁכֹּל יְחִידָה תוֹרֶמֶת לְכִסוּי הָעֲלוּת הַקְבוּעָה.

ש: וְהַדֻגְמָה הִיא מַאֲפִיָה.

א: חֲטִיף זֵיתִים. הָעֲלוּת הַמִשְׁתַנָה לַיְחִידָה: שְׁלוֹשִׁים וּשְׁתַיִם אֲגוֹרוֹת חוֹמְרֵי גֶלֶם, וַחֲמִשִׁים אֲגוֹרוֹת שְׂכַר עֲבוֹדָה. יַחַד, שְׁמוֹנִים וּשְׁתַיִם אֲגוֹרוֹת. הַמְחִיר, שִׁבְעָה שְׁקָלִים.
א: וְהָעֲלוּת הַקְבוּעָה? הַחֲטִיף מֵבִיא שְׁנֵי אֲחוּזִים מֵהַהַכְנָסוֹת שֶׁל הַמַאֲפִיָה, וְלָכֵן מְיַחֲסִים לוֹ שְׁנֵי אֲחוּזִים מֵהַהוֹצָאוֹת הַקְבוּעוֹת שֶׁלָה. יוֹצֵא עֶשְׂרִים וְאֶחָד אֶלֶף שֵׁשׁ מֵאוֹת אַרְבָּעִים וּשְׁנָיִם שֶׁקֶל.
א: נְקֻדַת הָאִזוּן: עֶשְׂרִים וְאֶחָד אֶלֶף שֵׁשׁ מֵאוֹת אַרְבָּעִים וּשְׁתַיִם, חֶלְקֵי שֵׁשׁ נְקֻדָה שְׁמוֹנֶה עֶשְׂרֵה. שְׁלֹשֶׁת אֲלָפִים חֲמֵשׁ מֵאוֹת וּשְׁתַיִם יְחִידוֹת. וְהַבִּקוּשׁ בְּפֹעַל הוּא שְׁלֹשֶׁת אֲלָפִים וּשְׁמוֹנֶה מֵאוֹת, מִיָמִין לִנְקֻדַת הָאִזוּן, בְּאֵזוֹר הָרֶוַח. רֶוַח שֶׁל אֶלֶף שְׁמוֹנָה מֵאוֹת אַרְבָּעִים וּשְׁנַיִם שֶׁקֶל.

ש: וְאָז נִתוּחֵי רְגִישׁוּת.

א: שְׁלוֹשָׁה תַרְחִישִׁים. מְחִיר נָמוּךְ בְּשִׁבְעָה אֲחוּזִים: נְקֻדַת הָאִזוּן עוֹלֶה לִשְׁלֹשֶׁת אֲלָפִים שְׁמוֹנֶה מֵאוֹת וְשָׁלוֹשׁ, וְהָרֶוַח הוֹפֵךְ לְהֶפְסֵד קָטָן, עֶשְׂרִים שֶׁקֶל. עֲלוּת קְבוּעָה גְבוֹהָה בַּחֲמִשָׁה אֲחוּזִים: עֲדַיִן רֶוַח, שְׁבַע מֵאוֹת וְשִׁשִׁים. וְהִתְיַקְרוּת שֶׁל חוֹמְרֵי הַגֶלֶם וּשְׂכַר הָעֲבוֹדָה: עֲדַיִן רֶוַח, אֶלֶף שֵׁשׁ מֵאוֹת אַרְבָּעִים וּשְׁמֹנָה.
א: וְהַמַסְקָנָה שֶׁל הַמַצֶגֶת: הוֹרָדַת מְחִיר, אֲפִלוּ קְטַנָה, יוֹצֶרֶת הֶפְסֵד. וְלָכֵן צָרִיךְ לְהִזָהֵר עִם הוֹרָדַת מְחִירִים.

ש: וּבְחִירָה בֵּין שְׁתֵי מְכוֹנוֹת?

א: מְכוֹנָה אָלֶף עוֹלֶה שֵׁשׁ מֵאוֹת אֶלֶף שֶׁקֶל, וַעֲלוּת הַיִצוּר בָּה עֶשְׂרִים וַחֲמִשָׁה שֶׁקֶל לִיחִידָה. מְכוֹנָה בֵּית עוֹלֶה אַרְבַּע מֵאוֹת וַחֲמִשִׁים אֶלֶף, וַעֲלוּת הַיִצוּר בָּה שְׁלוֹשִׁים וּשְׁנַיִם. מַשְׁוִים אֶת הַהוֹצָאוֹת: הַהֶפְרֵשׁ בַּהַשְׁקָעָה, מֵאָה וַחֲמִשִׁים אֶלֶף, חֶלְקֵי הַהֶפְרֵשׁ בַּעֲלוּת לִיחִידָה, שִׁבְעָה. יוֹצֵא עֶשְׂרִים וְאֶחָד אֶלֶף אַרְבַּע מֵאוֹת עֶשְׂרִים וְתֵשַׁע יְחִידוֹת.
א: הַפְּרוֹיֵקְט נִמְשַׁךְ חָמֵשׁ שָׁנִים, עִם חֲמֵשֶׁת אֲלָפִים יְחִידוֹת בְּשָׁנָה. כְּלוֹמַר עֶשְׂרִים וַחֲמִשָׁה אֶלֶף יְחִידוֹת, יוֹתֵר מִנְקֻדַת שִׁיווּי הַמִשְׁקָל. וְלָכֵן בּוֹחֲרִים בִּמְכוֹנָה אָלֶף, עִם הַהַשְׁקָעָה הַגְבוֹהָה וַעֲלוּת הַיְחִידָה הַנְמוּכָה.

ש: וְהַמַלְכֹּדֶת?

א: לְהַשְׁווֹת לְכַמוּת שֶׁל כֹּל הַפְּרוֹיֵקְט, וְלֹא לְכַמוּת שֶׁל שָׁנָה אַחַת. בְּשָׁנָה אַחַת, חֲמֵשֶׁת אֲלָפִים יְחִידוֹת, מְכוֹנָה בֵּית הָיְתָה נִרְאֵית זוֹלָה יוֹתֵר.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: טִפּוּל מוֹנֵעַ לוֹקֵחַ אַרְבַּע שָׁעוֹת, תִקוּן עָשָׂר שָׁעוֹת, יֵשׁ מָאתַיִם שְׁעוֹת עֲבוֹדָה בְּחֹדֶשׁ, וּבְלִי טִפּוּלִים יֵשׁ חָמֵשׁ תַקָלוֹת. מָה מִסְפַּר הַטִפּוּלִים הָאוֹפְּטִימָלִי?

~7

א: הַלוֹגָרִיתְם הַטִבְעִי שֶׁל שְׁתֵים עֶשְׂרֵה וָחֵצִי: שְׁתַיִם נְקֻדָה חֲמִשִׁים וְשָׁלוֹשׁ. קֶצֶב הַטִפּוּלִים חֲמִשִׁים, וְקֶצֶב הַתִקוּנִים עֶשְׂרִים.

ש: שְׁנִיָה: מָה יִקְרֶה לְמִסְפַּר הַטִפּוּלִים הָאוֹפְּטִימָלִי אִם זְמַן הַתִקוּן יִתְקַצֵר?

~4

א: הוּא יִקְטַן. כֹּל תַקָלָה מַשְׁבִּיתָה פָּחוֹת זְמַן, וְלָכֵן פָּחוֹת מִשְׁתַלֵם לְהַשְׁקִיעַ בְּטִפּוּלִים מוֹנְעִים.

ש: שְׁלִישִׁית: עֲלוּת קְבוּעָה שְׁלוֹשִׁים אֶלֶף, מְחִיר שְׁתֵים עֶשְׂרֵה, וַעֲלוּת מִשְׁתַנָה אַרְבָּעָה וָחֵצִי. מָה נְקֻדַת הָאִזוּן?

~5

א: אַרְבַּעַת אֲלָפִים יְחִידוֹת. שְׁלוֹשִׁים אֶלֶף חֶלְקֵי שִׁבְעָה וָחֵצִי.

ש: רְבִיעִית: בְּנִתוּחַ הָרְגִישׁוּת שֶׁל חֲטִיף הַזֵיתִים, אֵיזֶה שִׁנוּי הָפַךְ אֶת הָרֶוַח לְהֶפְסֵד?

~4

א: יְרִידָה שֶׁל שִׁבְעָה אֲחוּזִים בַּמְחִיר.

ש: חֲמִישִׁית: מְכוֹנָה עִם הַשְׁקָעָה גְבוֹהָה וַעֲלוּת יִצוּר נְמוּכָה. מָתַי נַעֲדִיף אוֹתָה?

~4

א: כְּשֶׁהַכַּמוּת הַנִדְרֶשֶׁת לְאֹרֶךְ כֹּל הַפְּרוֹיֵקְט גְדוֹלָה מִכַּמוּת שִׁיווּי הַמִשְׁקָל.

ש: זֶהוּ לַפֶּרֶק הַזֶה, וְהוּא סוֹגֵר אֶת הַמוֹדֵלִים שֶׁל נִצָן. בַּפֶּרֶק הַבָּא עוֹבְרִים לִשְׁלֹמֹה: מִבְנֵה שַׁרְשֶׁרֶת הָאַסְפָּקָה, וְנִהוּל חֹמֶר.
