# title: שלמה — מבנה השרשרת וניהול חומר
# מצגות שא 1 ו-שא 2 של שלמה ארליך. כל עובדה כאן מופיעה בבנקים 1 ו-2 או בדף הנוסחאות.
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.
# ראשי תיבות לועזיים מנוקדים ביד (אֶם אָר פִּי, וִי אֶם אַיי) ובלי אותיות שימוש צמודות.

ש: פֶּרֶק שְׁבִיעִי, וְעוֹבְרִים לַחֵלֶק שֶׁל שְׁלֹמֹה אֶרְלִיךְ. כָּאן הַסִגְנוֹן אַחֵר: פָּחוֹת נֻסְחָאוֹת, יוֹתֵר מֻשָׂגִים, רְשִׁימוֹת וּמוֹדֵלִים.

א: וּבַאֲתַר הַתִרְגוּל, רֹב הַשְׁאֵלוֹת עַל הַחֹמֶר שֶׁל שְׁלֹמֹה בְּנוּיוֹת כָּךְ: תַרְחִישׁ קָצָר מֵאִרְגוּן, וְצָרִיךְ לְזַהוֹת לְאֵיזֶה מֻשָׂג הוּא שַׁיָךְ. אָז בַּפְּרָקִים הָאֵלֶה נִתְמַקֵד בְּהַבְחָנוֹת: מָה שַׁיָךְ לְמָה, וְאֵיפֹה קַל לְהִתְבַּלְבֵּל.

ש: נַתְחִיל מֵהַתְמוּנָה הַגְדוֹלָה. מָה הַתַהֲלִיכִים הַמֶרְכָּזִיִים בָּאִרְגוּן?

א: שְׁלֹמֹה מְזַהֶה שְׁלוֹשָׁה תַהֲלִיכֵי אָב. פִּתוּחַ: מִצֹרֶךְ שׁוּק, לְמוּצָר בַּר מְכִירָה. אַסְפָּקָה: מִדְרִישַׁת לָקוֹחַ, לְלָקוֹחַ מִשְׁתַמֵשׁ. וְשֵׁרוּת: מִפְּנִיַת לָקוֹחַ, לְלָקוֹחַ מְרֻצֶה.

ש: וְאֵיפֹה הַמַלְכֹּדֶת?

א: בַּסוֹף שֶׁל תַהֲלִיךְ הָאַסְפָּקָה. הוּא לֹא נִגְמָר כְּשֶׁהַמִשְׁלוֹחַ יוֹצֵא מֵהַמַחְסָן. הוּא נִגְמַר כְּשֶׁהַלָקוֹחַ מִשְׁתַמֵשׁ בַּמוּצָר. וּבַשֵׁרוּת, הַפֶּלֶט הוּא לָקוֹחַ מְרֻצֶה, וְלֹא רַק מַכְשִׁיר מְתֻקָן.

ש: וְאֵיךְ נִרְאֶה תַהֲלִיךְ בְּאֹפֶן כְּלָלִי?

א: קֶלֶט, תַהֲלִיךְ, פֶּלֶט. וּבְתוֹךְ הַתַהֲלִיךְ, מַעְגַל פִּי דִי סִי אֵיי: תִכְנוּן, בִּצוּעַ, בְּדִיקָה, וּפְעֻלָה. לְמָשָׁל, מְנַהֵל מַחְסָן מְתַכְנֵן שִׁפּוּר, מְנַסֶה אוֹתוֹ בְּמַחְסָן אֶחָד, בּוֹדֵק אֶת הַתוֹצָאוֹת, וּלְפִיהֶן מַחֲלִיט אִם לְהַטְמִיעַ בְּכֹל הַמַחְסָנִים.

ש: וְיֵשׁ מוֹדֵל עִם שֵׁם מַפְתִיעַ: מוֹדֵל הַכַּדוּרֶגֶל.

א: כֵּן. מִבְנֵה הָאִרְגוּן דוֹמֶה לְמַעֲרָךְ שֶׁל קְבוּצַת כַּדוּרֶגֶל, לְמָשָׁל אַרְבַּע, שָׁלוֹשׁ, שָׁלוֹשׁ. וְהַמַעֲרָךְ, כְּלוֹמַר חֲלֻקַת הַמַשְׁאַבִּים בֵּין הַפוּנְקְצִיוֹת, מִשְׁתַנֶה לְפִי הַשָׁלָב בַּמַחְזוֹר הַחַיִים. חֶבְרָה שֶׁעוֹבֶרֶת מִפִּתוּחַ לִצְמִיחָה מַעֲבִירָה מַשְׁאַבִּים מֵהַפִּתוּחַ לִמְכִירוֹת וּלְתִפְעוּל.

ש: וּמָה הָעֶקְרוֹנוֹת לִבְנִיַת מִבְנֶה אִרְגוּנִי?

א: הַמִבְנֶה הָאִרְגוּנִי תוֹמֵךְ בְּתַהֲלִיךְ הָאָב. שְׁבִירַת הַתַהֲלִיךְ. שְׁמוֹנִים אָחוּז מֵהַתַהֲלִיכִים. יְחִידוֹת יִעוּדִיוֹת לְתַהֲלִיכִים יִעוּדִיִים. וּמִבְנֵה הַפָּנִים שֶׁל כֹּל יְחִידָה, לְפִי מִבְנֵה הַלָקוֹחוֹת שֶׁלָה.

ש: שְׁבִירַת הַתַהֲלִיךְ נִשְׁמָע כְּמוֹ מַשֶׁהוּ רַע.

א: וְזוֹ בְּדִיוּק הַמַלְכֹּדֶת. אֵצֶל שְׁלֹמֹה, שְׁבִירַת הַתַהֲלִיךְ הִיא מֻשָׂג חִיוּבִי: חֲלֻקָה שֶׁל תַהֲלִיךְ הָאָב לִתְחוּמֵי אַחְרָיוּת בְּרוּרִים. וְדֻגְמָה לָעִקָרוֹן הָאַחֲרוֹן: מַחְלָקָה שֶׁמְאַרְגֶנֶת אֶת הַצְוָתִים שֶׁלָה לְפִי סוּגֵי הַלָקוֹחוֹת, בָּתֵי חוֹלִים, רְשָׁתוֹת, וּמִכְרָזִים מֶמְשַׁלְתִיִים. זֶה מִבְנֵה פְּנִים לְפִי מִבְנֵה הַלָקוֹחוֹת.

ש: וְשַׁרְשֶׁרֶת פִּתוּחַ הַמוּצָר?

א: שִׁבְעָה שְׁלַבִּים, וְהַסֵדֶר חָשׁוּב. פִּתוּחַ עִסְקִי, שֶׁמַגְדִיר אֶת הַדְרִישָׁה. מֶחְקָר וּפִתוּחַ, שֶׁמְיַצֵר אַב טִיפּוּס. הַנְדָסָה, שֶׁמְכִינָה תִיק מוּצָר וְתִיק יִצוּר. תִפְעוּל, שֶׁמְיַצֵר אֶת הַמוּצָרִים. שִׁיווּק יוֹצֵא, שֶׁיוֹצֵר אֶת הַצֹרֶךְ. מְכִירוֹת, שֶׁמְבִיאוֹת לִרְכִישָׁה. וְשֵׁרוּת, לָקוֹחַ מְרֻצֶה.

ש: וּמִי נִמְצָא תַחַת הַתִפְעוּל?

א: הַנְדָסָה, שַׁרְשֶׁרֶת אַסְפָּקָה, שֵׁרוּת, רֶכֶשׁ וְאַדְמִינִיסְטְרַצְיָה. מְכִירוֹת לֹא. הֵן בַּאֲגַף הַשִׁיווּק וְהַמְכִירוֹת.
א: וּלְחֵלֶק מֵהַיְחִידוֹת יֵשׁ חֲלוּפוֹת מִקוּם. הַנְדָסָה יְכוֹלָה לִהְיוֹת בְּפִתוּחַ, אוֹ בְּתִפְעוּל כְּהַנְדָסַת יִצוּר. שֵׁרוּת, בְּשִׁיווּק אוֹ בְּתִפְעוּל. וְרֶכֶשׁ, בִּכְסָפִים אוֹ בְּתִפְעוּל, עִם הַבְחָנָה בֵּין רֶכֶשׁ טַקְטִי לְרֶכֶשׁ אַסְטְרָטֵגִי.

ש: עַכְשָׁו לְשַׁרְשֶׁרֶת הָאַסְפָּקָה עַצְמָה.

א: שְׁלֹמֹה מַבְחִין בֵּין שְׁתֵי שַׁרְשְׁרָאוֹת. הַחוּץ אִרְגוּנִית: מָקוֹר, סַפָּק, הָאִרְגוּן, לָקוֹחַ, וּמִשְׁתַמֵשׁ סוֹפִי. וְהַפְּנִים אִרְגוּנִית, שֶׁבְּנוּיָה מֵחֲמִשָׁה בְּלוֹקִים.

ש: מָה הֵם?

א: נִהוּל חֹמֶר: תַחֲזִיוֹת, אֶם אָר פִּי, וּמְלַאי. רֶכֶשׁ: מִקוּר, כְּלוֹמַר אִתוּר סַפָּקִים וּבְחִירָתָם, רֶכֶשׁ בְּפֹעַל, וְקִדוּם רֶכֶשׁ. יִצוּר: יִצוּר עַצְמִי, וְקַבְּלָנֵי מִשְׁנֶה. נִהוּל הַזְמָנוֹת: קִדוּם מִשְׁלוֹחִים. וְלוֹגִיסְטִיקָה: מַחְסָנִים, שִׁלוּחַ וַהֲפָצָה.

ש: וּמָה נָפוֹץ לְבַלְבֵּל?

א: שְׁלוֹשָׁה דְבָרִים. שֵׁרוּת הוּא לֹא בְּלוֹק בְּשַׁרְשֶׁרֶת. הוּא תַהֲלִיךְ אָב, וִיחִידָה בְּתִפְעוּל. יִצוּר אֵצֶל יַצְרָן חִיצוֹנִי נִשְׁאַר בִּבְּלוֹק הַיִצוּר, כְּקַבְּלָנוּת מִשְׁנֶה, וְלֹא עוֹבֵר לְרֶכֶשׁ. וְקִדוּם מִשְׁלוֹחִים לְלָקוֹחוֹת שַׁיָךְ לְנִהוּל הַהַזְמָנוֹת, בְּעוֹד שֶׁקִדוּם רֶכֶשׁ הוּא זֵרוּז אַסְפָּקוֹת מֵהַסַפָּקִים.

ש: מָה הַיִעוּד שֶׁל שַׁרְשֶׁרֶת הָאַסְפָּקָה?

א: לְהַבְטִיחַ אַסְפָּקוֹת לְלָקוֹחוֹת. בִּזְמַן, בְּאֵיכוּת גְבוֹהָה, בַּעֲלוּת מֵיטָבִית, וּבִזְמִישׁוּת, כְּלוֹמַר גְמִישׁוּת וִיכֹלֶת לִגְדֹל.

ש: וְהָאֶתְגָר הָעִקָרִי?

א: לְהַבְטִיחַ אַסְפָּקוֹת, תוֹךְ שְׁמִירָה עַל זְמִישׁוּת תִפְעוּלִית וְעַל רָמַת מְלַאי נְמוּכָה. שְׁלוֹשָׁה יְעָדִים שֶׁמוֹשְׁכִים לְכִיווּנִים שׁוֹנִים. וְשִׂימוּ לֵב: מְלַאי נָמוּךְ מוֹפִיעַ בָּאֶתְגָר, וְלֹא בַּיִעוּד.

ש: וּמָה עִסוּקֵי הַלִבָּה?

א: נִהוּל הַזְמָנוֹת וְאַסְפָּקוֹת, נִהוּל חֹמֶר וְתַחֲזִיוֹת, נִהוּל מְלַאי, נִהוּל סַפָּקִים וּמִקוּר, הוֹרָדַת עֲלוּיוֹת, לוֹגִיסְטִיקָה, וְשִׁפּוּר תַהֲלִיכִים וּבַקָרַת בִּצוּעִים. וְעוֹד אֶחָד, אֶן פִּי אַיי: תְמִיכַת הַתִפְעוּל בְּמוּצָרִים חֲדָשִׁים, הַמַעֲבָר מִפִּתוּחַ לְיִצוּר.

ש: וְאֵיךְ מוֹדְדִים אֶת כֹּל זֶה?

א: בְּעֵץ מְדָדִים. בַּשֹׁרֶשׁ שֶׁלוֹ, קוֹל הַלָקוֹחַ: אַסְפָּקָה לְלָקוֹחוֹת. וּמִתַחְתָיו הָעֲנָפִים. אֵיכוּת, עִם מוּצָרִים שֶׁהִגִיעוּ לֹא תַקִינִים, וְהַחֲזָרוֹת מִלָקוֹחוֹת. מְלַאי, עִם רָמַת הַמְלַאי, מַחְזוֹרֵי הַמְלַאי, וּמְלַאי מֵת אוֹ אִטִי. דִיוּק הַתַחֲזִית, כְּשֶׁהַתַחֲזִית נְמוּכָה מֵהַבִּקוּשׁ בְּפֹעַל, אוֹ גְבוֹהָה מִמֶנוּ. זְמַנֵי אַסְפָּקָה. עֲמִידָה בַּזְמַנִים, עִם מְמֻצַע יְמֵי הָאִחוּר וּמִסְפַּר הָאִחוּרִים. הַכְנָסוֹת שֶׁנִפְגְעוּ, עִם הַזְמָנוֹת שֶׁהִתְעַכְּבוּ וְהַזְמָנוֹת שֶׁאָבְדוּ. עֲלוּת תִפְעוּלִית. הוֹרָדַת עֲלוּיוֹת. וַעֲלוּת הַלוֹגִיסְטִיקָה בְּיַחַס לַעֲלוּת הַמֶכֶר.

ש: הַרְבֵּה עֲנָפִים. יֵשׁ מַלְכֹּדֶת?

א: מִסְפַּר הַסַפָּקִים. הוּא נִמְצָא בַּעֲנַף הוֹרָדַת הָעֲלוּיוֹת, וְלֹא בָּעֲלוּת הַתִפְעוּלִית. וְהַזְמָנָה שֶׁאָבְדָה בִּגְלַל חֹסֶר בַּמְלַאי שַׁיֶכֶת לַעֲנַף הַהַכְנָסוֹת שֶׁנִפְגְעוּ, וְלֹא לַעֲנַף הַמְלַאי.

ש: עוֹבְרִים לַמַצֶגֶת הַשְׁנִיָה, נִהוּל חֹמֶר. נַתְחִיל מִתַהֲלִיךְ אֶם אָר פִּי.

א: הַתַהֲלִיךְ מַתְחִיל מֵהַזְמָנוֹת וְתַחֲזִיוֹת. מֵהֶן בּוֹנִים תֹכְנִית יִצוּר אָב. מְפוֹצְצִים אֶת עֵץ הַמוּצָר, וּמְקַבְּלִים דְרִישׁוֹת בְּרוּטוֹ. מְקַזְזִים אֶת הַמְלַאי הַקַיָם וְאֶת הַהַזְמָנוֹת הַפְּתוּחוֹת, וּמְקַבְּלִים דְרִישׁוֹת נֵטוֹ. וְעַל דְרִישׁוֹת הַנֵטוֹ מוֹצִיאִים הַזְמָנוֹת רֶכֶשׁ וּפְקֻדוֹת עֲבוֹדָה.

ש: דֻגְמָה?

א: דְרִישַׁת בְּרוּטוֹ שֶׁל אֶלֶף יְחִידוֹת. בַּמַחְסָן יֵשׁ שָׁלוֹשׁ מֵאוֹת, וְיֵשׁ הַזְמָנַת רֶכֶשׁ פְּתוּחָה שֶׁל מָאתַיִם שֶׁעוֹד לֹא הִגִיעָה. דְרִישַׁת הַנֵטוֹ: אֶלֶף, פָּחוֹת שְׁלוֹשׁ מֵאוֹת, פָּחוֹת מָאתַיִם. חֲמֵשׁ מֵאוֹת.
א: וְהַהַזְמָנָה הַפְּתוּחָה מְקֻזֶזֶת, כִּי הִיא אַסְפָּקָה עֲתִידִית. מִי שֶׁמַזְמִין שְׁבַע מֵאוֹת, יְקַבֵּל מְלַאי עוֹדֵף.

ש: וּמָה עִם רְכִיבִים שֶׁזְמַן הָאַסְפָּקָה שֶׁלָהֶם אָרֹךְ בִּמְיֻחָד?

א: אֵלֶה פְּרִיטֵי אֶל אֶל אַיי. לְמָשָׁל רְכִיב שֶׁמַגִיעַ אַחֲרֵי אַרְבָּעִים שָׁבוּעוֹת, כְּשֶׁהַתַחֲזִית שֶׁל הַלָקוֹחוֹת הִיא לְרִבְעוֹן בִּלְבַד. צָרִיךְ לְהִתְחַיֵב עֲלֵיהֶם מֵעֵבֶר לָאֹפֶק הָרָגִיל, וְלָכֵן הֵם מְקַבְּלִים הִתְיַחֲסוּת נִפְרֶדֶת.

ש: וְאֵלוּ סוּגֵי מְלַאי יֵשׁ?

א: לְכֹל סוּג יֵשׁ מַטָרָה. חוֹמְרֵי גֶלֶם: זְמִינוּת לְיִצוּר, וְאִי תְלוּת בַּסַפָּקִים. מְלַאי בַּתַהֲלִיךְ, בֵּין הַתַחֲנוֹת: שֶׁטֶף הַיִצוּר, וְאִי תְלוּת בֵּין הַתַחֲנוֹת. תוֹצֶרֶת גְמוּרָה: זְמִינוּת הָאַסְפָּקוֹת, וְאִי תְלוּת שֶׁל הָאַסְפָּקוֹת בַּיִצוּר.

ש: וּמְלַאי מְנֹהָל סַפָּק?

א: וִי אֶם אַיי. מְלַאי זָמִין לְשִׁמוּשׁ, שֶׁיֵשׁ עָלָיו מְחֻיָבוּת. לְעֻמַת קוֹנְסִיגְנַצְיָה: מְלַאי אֵצֶל הַלָקוֹחַ, שֶׁהַלָקוֹחַ לֹא מְחֻיָב עָלָיו.
א: וְיֵשׁ עוֹד שְׁנַיִם שֶׁמִתְבַּלְבְּלִים בֵּינֵיהֶם. אֶם אָר בִּי: מוּצָרִים תְקוּלִים שֶׁחָזְרוּ מִלָקוֹחוֹת. וְהַשֵׁנִי, אָר טִי וִי: מוּצָרִים שֶׁנִמְצָאִים בְּתַהֲלִיךְ תִקוּן.

ש: וְעֵץ הַמְלַאי?

א: הוּא מְחַלֵק אֶת הַמְלַאי לִשְׁנֵי עֲנָפִים. הַמְלַאי הַתִפְעוּלִי: מְלַאי אֵצֶל לָקוֹחוֹת, תוֹצֶרֶת גְמוּרָה, חוֹמְרֵי גֶלֶם וּמְלַאי בְּתַהֲלִיךְ, מְלַאי שֵׁרוּת, תְקוּלִים, בְּתִקוּן, וּמְלַאי מְתֻקָן. וְהַמְלַאי הַלֹא תִפְעוּלִי: מְלַאי אִטִי אוֹ מֵת, מְלַאי שֶׁל פִּתוּחַ וְהַנְדָסָה, וּרְכוּשׁ קָבוּעַ.

ש: כְּלוֹמַר חֶלְקֵי חִלוּף לְרֶכֶב שֶׁיָצָא מִשֵׁרוּת לִפְנֵי שָׁנִים?

א: מְלַאי מֵת, בֶּעָנָף הַלֹא תִפְעוּלִי. וּמַכְשִׁיר שֶׁחָזַר מִלָקוֹחַ, תֻקַן, וּמְחַכֶּה לְהִשָׁלַח שׁוּב? מְלַאי מְתֻקָן, וְהוּא דווקא תִפְעוּלִי.

ש: וּמְדִינִיוּת הַמְלַאי?

א: גַם שְׁלֹמֹה מִשְׁתַמֵשׁ בַּחֲלֻקַת אֵיי בִּי סִי, אֲבָל עִם מִסְפָּרִים קְצָת אֲחֵרִים מִשֶׁל נִצָן. אֵצֶל שְׁלֹמֹה, קְבוּצָה אֵיי הִיא כַּעֲשָׂרָה אֲחוּזִים מֵהַכַּמוּת, וּשְׁמוֹנִים אָחוּז מֵהָעֵרֶךְ. וְהַמְדִינִיוּת: נִהוּל וּבַקָרָה רְצִיפִים, וְהַזְמָנָה לְפִי מוֹדֵל הַכַּמוּת הַכַּלְכָּלִית, אוֹ לְפִי אֶם אָר פִּי.
א: קְבוּצָה בִּי, כִּשְׁלוֹשִׁים אָחוּז מֵהַכַּמוּת, וַחֲמִשָׁה עָשָׂר אָחוּז מֵהָעֵרֶךְ. וּקְבוּצָה סִי, כְּשִׁשִׁים אָחוּז מֵהַכַּמוּת, וַחֲמִשָׁה אֲחוּזִים מֵהָעֵרֶךְ. בָּה קוֹנִים מָנָה שֶׁמַסְפִּיקָה לִתְקוּפָה אֲרֻכָּה, בְּלִי בַּקָרָה צְמוּדָה.

ש: וְנִצָן?

א: אֵצֶל נִצָן, חֲמִשָׁה אֲחוּזִים מֵהַפְּרִיטִים הֵם שְׁמוֹנִים אָחוּז מֵהָעֵרֶךְ, וּשְׁמוֹנִים אָחוּז מֵהַפְּרִיטִים הֵם חֲמִשָׁה אֲחוּזִים מֵהָעֵרֶךְ. בִּשְׁנֵי הַמִקְרִים, קְבוּצָה אֵיי הִיא כִּשְׁמוֹנִים אָחוּז מֵהָעֵרֶךְ. אֲבָל אִם שׁוֹאֲלִים עַל אֲחוּז הַפְּרִיטִים, כְּדַאי לָשִׂים לֵב שֶׁל מִי הַמַצֶגֶת.

ש: וּמַטְרִיצַת הַגְמִישׁוּת?

א: הֶסְכֵּם בֵּין לָקוֹחַ לַמִפְעָל: כַּמָה אֶפְשָׁר לְשַׁנוֹת הַזְמָנָה, וּמָתַי. בַּחֹדֶשׁ הַנוֹכְחִי, אֵין שִׁנוּי. לַחֹדֶשׁ הַבָּא, אֶפְשָׁר לְהַגְדִיל עַד עֲשָׂרָה אֲחוּזִים, אוֹ לְהַקְטִין עַד חֲמִשָׁה עָשָׂר. בְּעוֹד חוֹדְשַׁיִם, הַגְדָלָה עַד עֶשְׂרִים, וְהַקְטַנָה עַד עֶשְׂרִים וַחֲמִשָׁה. בְּעוֹד שְׁלוֹשָׁה חוֹדָשִׁים, הַגְדָלָה עַד שְׁלוֹשִׁים, וְהַקְטַנָה עַד שְׁלוֹשִׁים וַחֲמִשָׁה. וּמֵעֵבֶר לִשְׁלוֹשָׁה חוֹדָשִׁים, אֵין מְחֻיָבוּת.

ש: טְרִיק לִזְכִירָה: הַהַקְטָנָה תָמִיד גְדוֹלָה בְּחָמֵשׁ נְקֻדוֹת אָחוּז מֵהַהַגְדָלָה.

א: נָכוֹן. וְדֻגְמָה: לָקוֹחַ מְבַקֵשׁ לְהַגְדִיל בְּעֶשְׂרִים וַחֲמִשָׁה אָחוּז אֶת הַהַזְמָנָה לְעוֹד חוֹדְשַׁיִם. הַמִפְעָל מְחֻיָב רַק לְעֶשְׂרִים. וְלָקוֹחַ שֶׁמְבַקֵשׁ לְבַטֵל שְׁלוֹשִׁים אָחוּז מֵהַהַזְמָנָה לַחֹדֶשׁ הַבָּא? הַמִפְעָל מְחֻיָב לְקַבֵּל הַקְטָנָה שֶׁל חֲמִשָׁה עָשָׂר אָחוּז בִּלְבַד.

ש: וּמָה עוֹלֶה מְלַאי?

א: אַרְבָּעָה מַרְכִּיבִים. עֲלוּת רְכִישָׁה, כּוֹלֵל הֲנָחוֹת כַּמוּת. עֲלוּת הַזְמָנָה: מִשְׁלוֹחַ, וְנִהוּל הַהַזְמָנוֹת. עֲלוּת אַחְזָקָה: עֲלוּת הַהוֹן, וְהַפַּחַת. וַעֲלוּת חֹסֶר: דְחִיָה וְאוֹבְדַן שֶׁל אַסְפָּקוֹת.
א: וְעֵרֶךְ הַמְלַאי נִקְבַּע לְפִי הַנָמוּךְ מִבֵּין הָעֲלוּת לְבֵין מְחִיר הַשׁוּק. קָנִית בְּמֵאָה, וְהַשׁוּק יָרַד לְשִׁבְעִים? הַמְלַאי שָׁוֶה שִׁבְעִים. וְהַשִׁיטוֹת לְחִשׁוּב עֵרֶךְ הַמְלַאי: מְחִיר אַחֲרוֹן, מְמֻצָע נָע, פִיפוֹ, וְהַעֲרָכָה מֵחָדָשׁ. שִׁיטַת לִיפוֹ לֹא מוֹפִיעָה בָּרְשִׁימָה.

ש: וּבְעַד וְנֶגֶד מְלַאי?

א: בְּעַד: הֲגַנָה מֵעֲלִיַת מְחִירִים, וְצִמְצוּם עֲלוּיוֹת הַהַזְמָנָה. יְכֹלֶת תַחֲרוּת בְּאַסְפָּקָה מִיָדִית. וּבְמוּצָר עוֹנָתִי, יִצוּר מְאֻזָן לְאֹרֶךְ הַשָׁנָה, לְמָשָׁל מַזְגָנִים שֶׁמְיֻצָרִים בַּחֹרֶף לִקְרַאת הַקַיִץ.
א: וְנֶגֶד: עֲלוּת הַהוֹן, הָאִחְסוּן, הַפַּחַת, וְהַהִתְיַשְׁנוּת. וְהַהִתְיַשְׁנוּת יְכוֹלָה לִהְיוֹת פִיזִית, טֶכְנוֹלוֹגִית, אוֹ אוֹפְנָתִית. כִּסוּי לַטֵלֵפוֹן שֶׁהַדֶגֶם שֶׁלוֹ הֻחְלַף? הִתְיַשְׁנוּת טֶכְנוֹלוֹגִית.

ש: וְאֵיךְ שׁוֹמְרִים עַל זְמִישׁוּת, בְּלִי לְהַגְדִיל מְלַאי בְּכֹל מָקוֹם?

א: עִם קַוֵי הֲגַנָה מְמֻקָדִים. מְלַאי תוֹצֶרֶת גְמוּרָה. בַּפֶר בְּתַחֲזִית הַיִצוּר, כְּלוֹמַר תוֹסֶפֶת קְבוּעָה לַתַחֲזִית. בַּפֶר שֶׁל חוֹמְרֵי גֶלֶם. מְלַאי קוֹנְסִיגְנַצְיָה אֵצֶל הַיַצְרָן. וּמְלַאי מְנֹהָל סַפָּק, אֵצֶל הַיַצְרָן אוֹ אֵצֶל הַסַפָּק.

ש: וְזֶה מֵבִיא אוֹתָנוּ לְתִכְנוּן רָזֶה.

א: שִׁיטָה נִהוּלִית לְשִׁפּוּר מַתְמִיד שֶׁל תִכְנוּן הַחֹמֶר, כְּדֵי לְהַבְטִיחַ אַסְפָּקוֹת וּלְצַמְצֵם מְלַאי. וְהַמִשְׁפָּט שֶׁשְׁלֹמֹה חוֹזֵר עָלָיו: הַגְדָלַת הַמְלַאי אֵינָה הַפִּתְרוֹן.
א: הָעֶקְרוֹנוֹת, בִּקְצָרָה. מְשִׁיכָה: יִצוּר לְפִי דְרִישַׁת הַלָקוֹחַ, וְלֹא לְפִי תֹכְנִית קְבוּעָה. הַצָעַת עֵרֶךְ. בִּטוּל בִּזְבּוּז. זְמִישׁוּת. שִׁתוּף מֵידָע, לְמָשָׁל לִשְׁלֹחַ לַסַפָּקִים כֹּל שָׁבוּעַ אֶת הַתַחֲזִית, הַמְלָאִים וְתֹכְנִית הַיִצוּר. דִיוּק. שִׁפּוּר מַתְמִיד. שֻׁתָפוּת. וְהוֹן אֱנוֹשִׁי.

ש: וּמָה עוֹד כְּדַאי לִזְכֹּר מֵהַחֵלֶק הַזֶה?

א: שְׁלוֹשָׁה מֻשָׂגִים. מְלַאי זָמִין לִמְכִירָה: כַּמָה מֵהַמְלַאי הַקַיָם אֶפְשָׁר לְהַבְטִיחַ לְלָקוֹחוֹת חֲדָשִׁים. תִכְנוּן לְפִי מְכִירוֹת אֲמִתִיוֹת, וְלֹא לְפִי תַחֲזִית אוֹפְּטִימִית. וְלִסְגֹר אֶת הַבֶּרֶז, וְלֹא אֶת הַדְלִיפָה: בִּמְקוֹם לְפַנוֹת עֳדָפִים, לִבְדֹק לָמָה נִכְנַס לַמְלַאי יוֹתֵר מִדַי, וְלַעֲצֹר אֶת זֶה בַּמָקוֹר.

ש: וְנִשְׁאֲרוּ מוֹדֵלֵי הַטִיווּחַ. שֵׁמוֹת מֵהַתְחוּם הַצְבָאִי.

א: אַרְבָּעָה מוֹדֵלִים, לְפִי כַּמָה יוֹדְעִים עַל הַמַטָרָה, כְּלוֹמַר עַל הַבִּקוּשׁ. הַצַלָף: מַטָרָה בְּרוּרָה וְנַיַחַת, בִּקוּשׁ קָבוּעַ. הַפּוֹגֵשׁ: הַמַטָרָה זָזָה בְּכִיווּן יָדוּעַ, וּמְיָרְטִים אוֹתָה בַּמָקוֹם שֶׁבּוֹ הִיא תִהְיֶה. לְמָשָׁל בִּקוּשׁ שֶׁגָדַל בְּקֶצֶב קָבוּעַ, וּמְתַכְנְנִים לְרָמָה שֶׁתִהְיֶה כְּשֶׁהַהַזְמָנָה תַגִיעַ. וְהָרוֹדֵף: הַמַטָרָה זָזָה בְּכִיווּן לֹא בָּרוּר, וְעוֹקְבִים אַחֲרֶיהָ וּמְתַקְנִים כֹּל הַזְמַן.

ש: וְהָרְבִיעִי?

א: הַצַלָף הָעִוֵר. אֵין מֵידָע בִּכְלָל, לְמָשָׁל מוּצָר חָדָשׁ לְגַמְרֵי. וְאָז מְבִיאִים אֶת הַמַטָרָה לַמָקוֹם שֶׁמְכַוְנִים אֵלָיו. יוֹרִים לְמָקוֹם שֶׁבּוֹ הַמַטָרָה הָיְתָה, כְּלוֹמַר תַחֲזִית עָבָר. אוֹסְפִים מוֹדִיעִין. מַלְכֹּדֶת וּפִתָיוֹן, לְמָשָׁל מִבְצַע הַשָׁקָה שֶׁמְכַוֵן אֶת הַלָקוֹחוֹת לִקְנוֹת בְּדִיוּק אֶת מָה שֶׁיוֹצֵר. אוֹ קֻפְסַת אֵשׁ: יוֹרִים לְכֹל עֵבֶר.

ש: וּמָה עוֹשִׂים כְּשֶׁאֵין תַחֲזִית?

א: הַמַצֶגֶת מַצִיעָה תִכְנוּן לְלֹא תַחֲזִית. מִתְבַּסְסִים עַל נְתוּנִים וְהִיסְטוֹרְיָה, מַבְטִיחִים גְמִישׁוּת, מְקַצְרִים זְמַנֵי אַסְפָּקָה, עוֹבְרִים לְיִצוּר בִּיחִידָה בּוֹדֶדֶת בִּמְקוֹם בְּמָנוֹת גְדוֹלוֹת, וּמְתַכְנְנִים בְּצוּרָה טֵלֶסְקוֹפִּית וּפְרוֹאַקְטִיבִית.

ש: וּשְׁתֵי אֲמִירוֹת שֶׁל שְׁלֹמֹה לְסִיוּם.

א: לָתֵת חַכּוֹת וְלֹא דָגִים: לָתֵת לַצֶוֶת כֵּלִים, וְלֹא פִּתְרוֹנוֹת מוּכָנִים. וּלְשַׂחֵק מַטְקוֹת וְלֹא טֵנִיס: מוּל הַסַפָּקִים מְשַׁתְפִים פְּעֻלָה, וְלֹא מִתְחָרִים.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: מָתַי מִסְתַיֵם תַהֲלִיךְ הָאַסְפָּקָה?

~4

א: כְּשֶׁהַלָקוֹחַ מִשְׁתַמֵשׁ בַּמוּצָר. מִדְרִישַׁת לָקוֹחַ, לְלָקוֹחַ מִשְׁתַמֵשׁ.

ש: שְׁנִיָה: דְרִישַׁת בְּרוּטוֹ שֶׁל שְׁמוֹנֶה מֵאוֹת, מְלַאי שֶׁל מָאתַיִם וַחֲמִשִׁים, וְהַזְמָנָה פְּתוּחָה שֶׁל מֵאָה וַחֲמִשִׁים. מָה דְרִישַׁת הַנֵטוֹ?

~5

א: אַרְבַּע מֵאוֹת. שְׁמוֹנָה מֵאוֹת, פָּחוֹת מָאתַיִם וַחֲמִשִׁים, פָּחוֹת מֵאָה וַחֲמִשִׁים.

ש: שְׁלִישִׁית: לָקוֹחַ מְבַקֵשׁ לְהַגְדִיל בַּחֲמִשָׁה עָשָׂר אָחוּז אֶת הַהַזְמָנָה לַחֹדֶשׁ הַבָּא. לָמָה הַמִפְעָל מְחֻיָב?

~5

א: לַעֲשָׂרָה אֲחוּזִים בִּלְבַד. זוֹ הַגְמִישׁוּת הַמֻסְכֶּמֶת לְהַגְדָלָה בַּחֹדֶשׁ הַבָּא.

ש: רְבִיעִית: מוּצָרִים תְקוּלִים שֶׁחָזְרוּ מִלָקוֹחוֹת, וּמוּצָרִים שֶׁכְּבָר בְּתִקוּן. אֵיךְ נִקְרָאִים שְׁנֵי סוּגֵי הַמְלַאי?

~5

א: הָרִאשׁוֹנִים, אֶם אָר בִּי. הַשְׁנַיִם, אָר טִי וִי.

ש: חֲמִישִׁית: הַבִּקוּשׁ מִשְׁתַנֶה בְּלִי כִּיווּן בָּרוּר, אֲבָל הַמְתַכְנֵן רוֹאֶה אוֹתוֹ בִּזְמַן אֱמֶת וּמְתַקֵן. אֵיזֶה מוֹדֵל טִיווּחַ?

~4

א: הָרוֹדֵף.

ש: זֶהוּ לַפֶּרֶק הַזֶה. בַּפֶּרֶק הַבָּא: רֶכֶשׁ, יִצוּר, וּמִקוּר חוּץ. שְׁלַבֵּי תַהֲלִיךְ הָרֶכֶשׁ, מָתַי לְיַצֵר וּמָתַי לִקְנוֹת, וְהַפִּירָמִידָה שֶׁל מִקוּר הַחוּץ.
