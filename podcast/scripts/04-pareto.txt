# title: פארטו ופילוחים — אי בי סי ופילוח רב ממדי
# מצגת 6 של ניצן. כל עובדה כאן מופיעה בבנק 16 או בדף הנוסחאות.
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.
# הקבוצות A, B, C נכתבות „אֵיי, בִּי, סִי״ ומנוקדות ביד.

ש: פֶּרֶק רְבִיעִי: פָּארֶטוֹ וּפִלוּחִים. נוֹשֵׂא שֶׁנִשְׁמַע פָּשׁוּט, וּבֶאֱמֶת הַחִשׁוּב פָּשׁוּט, אֲבָל יֵשׁ בּוֹ כַּמָה מַלְכּוֹדוֹת.

א: נַתְחִיל מֵהַסִפּוּר. וִילְפֶּרְדוֹ פָּארֶטוֹ הָיָה כַּלְכְּלָן אִיטַלְקִי, שֶׁיִעֵץ לַאֲגַף הַגְבִיָה שֶׁל מַס הַכְנָסָה. הוּא מָצָא בְּחִשׁוּב פָּשׁוּט שֶׁכִּשְׁמוֹנִים אָחוּז מֵהַהוֹן שֶׁל מִשְׁקֵי הַבַּיִת בְּאִיטַלְיָה נִמְצָא בִּידֵי כְּעֶשְׂרִים אָחוּז מֵהַמִשְׁפָּחוֹת. וְלָכֵן הִמְלִיץ לְהַשְׁקִיעַ אֶת רֹב מַאֲמַצֵי הַגְבִיָה בַּמִשְׁפָּחוֹת הָאֵלֶה.

ש: וְהַמַצֶגֶת שׁוֹאֶלֶת שְׁאֵלָה מַכְשִׁילָה.

א: כֵּן. הַאִם זֶה אוֹמֵר שֶׁתִשְׁעִים אָחוּז מֵהַהוֹן נִמְצָאִים בִּידֵי עֲשָׂרָה אֲחוּזִים מֵהַמִשְׁפָּחוֹת? כְּלוֹמַר, הַאִם שְׁנֵי הַמִסְפָּרִים חַיָבִים לְהַשְׁלִים לְמֵאָה? וְהַתְשׁוּבָה: לֹא. שְׁמוֹנִים הוּא חֵלֶק מֵהַהוֹן, וְעֶשְׂרִים הוּא חֵלֶק מֵהָאֻכְלוּסִיָה. שְׁנֵי בְּסִיסִים שׁוֹנִים.
א: וְהַדֻגְמָה מִיִשְׂרָאֵל מַמְחִישָׁה אֶת זֶה: כְּאַרְבָּעִים וַחֲמִשָׁה אָחוּז מֵהַהוֹן וְהַנְכָסִים נִמְצָאִים בִּידֵי רֶבַע אָחוּז מֵהָאֻכְלוּסִיָה. אַרְבָּעִים וְחָמֵשׁ וְעוֹד רֶבַע, רָחוֹק מְאוֹד מִמֵאָה.

ש: וְהַמַצֶגֶת מַרְאָה שֶׁפָּארֶטוֹ נִמְצָא בְּכֹל מָקוֹם.

א: נָכוֹן. בְּרֶשֶׁת שֶׁל מֵאָה סְנִיפֵי קֻפַּת חוֹלִים, רֹב הַתְלוּנוֹת מִתְרַכְּזוֹת בַּעֲשָׂרָה עַד חֲמִשָׁה עָשָׂר סְנִיפִים. הַהַמְלָצָה: לְמַקֵד אֶת מַאֲמַצֵי הַשִׁפּוּר בָּהֶם. רֹב תְאוּנוֹת הַדְרָכִים הַקַטְלָנִיוֹת מִתְרַחֲשׁוֹת בְּמִסְפָּר קָטָן שֶׁל כְּבִישִׁים, הַכְּבִישִׁים הָאֲדֻמִים. וְרֹב הַפְּנִיוֹת לְמִשְׂרָדִים מֶמְשַׁלְתִיִים מַגִיעוֹת מִמִסְפָּר קָטָן שֶׁל מִתְלוֹנְנִים סִדְרָתִיִים.

ש: וּמָה הַיִתְרוֹן שֶׁל הַשִׁיטָה?

א: הַמַצֶגֶת אוֹמֶרֶת אֶת זֶה בְּמִשְׁפָּט אֶחָד: חִשׁוּב פָּשׁוּט, מַסְקָנוֹת מַשְׁמָעוּתִיוֹת. וְאֶפְשָׁר לְפַלֵחַ כָּךְ כֹּל תְחוּם עִסְקִי.

ש: מָה שְׁלַבֵּי הָעֲבוֹדָה?

א: אַרְבָּעָה, וְהַסֵדֶר חָשׁוּב. אֶחָד: מַצִיגִים אֶת הַפְּרִיטִים עִם הָעֲרָכִים הַכַּמוּתִיִים שֶׁלָהֶם. שְׁתַיִם: מְדָרְגִים מֵהַגָבוֹהַ לְנָמוּךְ. שָׁלוֹשׁ: מְחַשְׁבִים אָחוּז יַחֲסִי וְאָחוּז מִצְטַבֵּר. וְאַרְבַּע: מְשַׂרְטְטִים גְרָף, וּמַסִיקִים מַסְקָנוֹת נִהוּלִיוֹת. אֶת הָאָחוּז הַמִצְטַבֵּר אֶפְשָׁר לַחְשֹׁב רַק אַחֲרֵי הַדֵרוּג.

ש: וְהַדֻגְמָה הָרִאשׁוֹנָה?

א: תְלוּנוֹת עַל שֵׁרוּת מִשְׁלוֹחֵי אֹכֶל בִּתְקוּפַת הַקוֹרוֹנָה. שִׁבְעִים וָשֵׁשׁ פְּנִיוֹת. זְמַן מִשְׁלוֹחַ אָרֹךְ, שְׁלוֹשִׁים וְחָמֵשׁ. אֹכֶל קַר, עֶשְׂרִים. הַזְמָנָה לֹא מְדֻיֶקֶת, שְׁלוֹשׁ עֶשְׂרֵה. אֲרִיזַת הַמִשְׁלוֹחַ, שֵׁשׁ. וְשָׁלִיחַ לֹא אָדִיב, שְׁתַיִם.
א: זְמַן מִשְׁלוֹחַ אָרֹךְ הוּא אַרְבָּעִים וְשִׁשָׁה אָחוּז מֵהַפְּנִיוֹת. יַחַד עִם אֹכֶל קַר, כְּבָר שִׁבְעִים וּשְׁנַיִם אָחוּז מִצְטַבֵּר. הַהַמְלָצָה הַנִהוּלִית: לְקַצֵר אֶת זְמַנֵי הַמִשְׁלוֹחַ, וְלִדְאֹג שֶׁהָאֹכֶל יֵצֵא חַם וְשֶׁהַהַזְמָנָה תִהְיֶה מְדֻיֶקֶת.

ש: וְעַכְשָׁו פִּלוּחַ רַב מְמַדִי. רֶשֶׁת מִרְפָּאוֹת.

א: שִׁבְעָה סְנִיפִים. אִם מְדָרְגִים לְפִי מִסְפַּר הַתְלוּנוֹת, סְנִיף חֵרוּת בָּרֹאשׁ, עִם שְׁמוֹנִים וְחָמֵשׁ תְלוּנוֹת בְּשָׁנָה. אֲבָל הַמַצֶגֶת שׁוֹאֶלֶת: לָמָה הַפִּלוּחַ הַזֶה לֹא אֵיכוּתִי?

ש: כִּי חֵרוּת הוּא גַם סְנִיף גָדוֹל.

א: בְּדִיוּק. יֵשׁ בּוֹ יוֹתֵר מֵחֲמֵשֶׁת אֲלָפִים מְבַקְרִים בְּחֹדֶשׁ. סְנִיף גָדוֹל יְקַבֵּל יוֹתֵר תְלוּנוֹת, גַם אִם הַשֵׁרוּת בּוֹ טוֹב. לָכֵן מְנַרְמְלִים: תְלוּנוֹת חֶלְקֵי מְבַקְרִים.
א: וְאָז הַתְמוּנָה מִתְהַפֶּכֶת. סְנִיף גִלְעָד, עִם שְׁלוֹשִׁים וְתֵשַׁע תְלוּנוֹת בִּלְבַד, מִתְגַלֶה כִּבְעָיָתִי: כִּשְׁמוֹנָה אֲחוּזִים מֵהַמְבַקְרִים שֶׁלוֹ מִתְלוֹנְנִים. וְחֵרוּת, חוֹפִית וְיַרְמוּךְ יוֹצְאִים סְנִיפִים מִצְטַיְנִים.

ש: וְאָז בּוֹדְקִים אִם הַבְּעָיָה בְּגִלְעָד הִיא עֹמֶס.

א: כֵּן. מְחַשְׁבִים מְבַקְרִים לְכֹל עוֹבֵד. וְיַרְמוּךְ, הַסְנִיף הֶעָמוּס בְּיוֹתֵר, כִּמְעַט בְּלִי תְלוּנוֹת בְּיַחַס לַמְבַקְרִים שֶׁלוֹ. הַמַסְקָנָה שֶׁל הַמַצֶגֶת: אֵין מִתְאָם בֵּין הָעֹמֶס לַתְלוּנוֹת, וְלָכֵן הַבְּעָיָה בְּגִלְעָד נוֹבַעַת כַּנִרְאֶה מִשֵׁרוּת לֹא אֵיכוּתִי.
א: וְעוֹד פִּלוּחַ אֶחָד, לְפִי נֵתַח שׁוּק. וְהוּא מַרְאֶה שֶׁהַלָקוֹחוֹת מַצְבִּיעִים בָּרַגְלַיִם: הֵם עוֹזְבִים סְנִיפִים עִם בְּעָיַת תְלוּנוֹת חֲמוּרָה, כְּמוֹ גִלְעָד, וּמִצְטָרְפִים לַסְנִיפִים עִם שֵׁרוּת טוֹב.

ש: וְזֶה נִקְרָא פִּלוּחַ רַב מְמַדִי.

א: כֵּן. כַּמָה פִּלוּחִים בַּחֲתָכִים שׁוֹנִים, שֶׁכֹּל אֶחָד מוֹסִיף נִדְבָּךְ. אַגַב, בְּטַבְלַת הָעֹמֶס בַּשֶׁקֶף שֵׁשׁ עֶשְׂרֵה, שְׁמוֹת הַסְנִיפִים הִתְחַלְפוּ לְעֻמַת טַבְלַת הַנְתוּנִים. הַמַסְקָנָה נְכוֹנָה, אֲבָל לֹא כְּדַאי לִסְמֹךְ עַל הַדֵרוּג שֶׁבַּשֶׁקֶף.

ש: עוֹבְרִים לַמְלַאי. אֵילוּ פִּלוּחִים אֶפְשָׁר לַעֲשׂוֹת לִפְרִיטֵי מְלַאי?

א: לְפִי עֵרֶךְ הַצְרִיכָה, לְפִי הַצְרִיכָה עַצְמָה, לְפִי רְגִישׁוּת לְפָג תֹקֶף, לְפִי נֶפַח, לְפִי אֹרֶךְ זְמַנֵי הָאַסְפָּקָה, לְפִי מֻרְכָּבוּת הָאַסְפָּקָה, וּלְפִי קְרִיטִיוֹת.

ש: וְהַפִּלוּחַ הַמֶרְכָּזִי הוּא לְפִי עֵרֶךְ הַצְרִיכָה הַשְׁנָתִית.

א: עֵרֶךְ הַצְרִיכָה הוּא הַצְרִיכָה הַשְׁנָתִית, כָּפוּל מְחִיר הַיְחִידָה. הוּא מְבַטֵא אֶת הַמִשְׁקָל הַכַּלְכָּלִי שֶׁל הַפְּרִיט, כְּלוֹמַר כַּמָה הוּא תוֹרֵם לְמַחְזוֹר הַכְּסָפִים. וּלְפִיו מְחַלְקִים אֶת הַפְּרִיטִים לְשָׁלוֹשׁ קְבוּצוֹת, כְּדֵי לְנַהֵל כֹּל קְבוּצָה בִּמְדִינִיוּת אַחֶרֶת.

ש: וְהַדֻגְמָה הִיא מוּסָךְ.

א: אַחַד עָשָׂר פְּרִיטִים. בָּרֹאשׁ: מָנוֹעַ, מֵאָה יְחִידוֹת בִּמְחִיר שֶׁל שִׁשָׁה אֲלָפִים, כְּלוֹמַר שֵׁשׁ מֵאוֹת אֶלֶף שֶׁקֶל. וְאַחֲרָיו גִיר, מֵאָה יְחִידוֹת בַּחֲמֵשֶׁת אֲלָפִים, חֲמֵשׁ מֵאוֹת אֶלֶף.
א: שְׁנֵי הַפְּרִיטִים הָאֵלֶה לְבַד הֵם כְּשִׁבְעִים וְתִשְׁעָה אָחוּז מֵעֵרֶךְ הַצְרִיכָה. הֵם קְבוּצָה אֵיי. שְׁלוֹשָׁה פְּרִיטִים נוֹסָפִים, מַשְׁאֵבַת מַיִם, צְמִיגִים וּפִילְטֵרִים, הֵם כְּעֶשְׂרִים אָחוּז, קְבוּצָה בִּי. וְשִׁשָׁה פְּרִיטִים שֶׁל בְּרָגִים, אוֹמִים וְדִסְקִיוֹת, פָּחוֹת מֵאָחוּז אֶחָד. קְבוּצָה סִי.

ש: שִׂימוּ לֵב שֶׁהַבְּרָגִים נִצְרָכִים בְּכַמוּיוֹת הֲכִי גְדוֹלוֹת.

א: נָכוֹן, וְזוֹ מַלְכֹּדֶת. הַדֵרוּג הוּא לְפִי הַמַכְפֵּלָה שֶׁל צְרִיכָה וּמְחִיר, וְלֹא לְפִי הַכַּמוּת, וְגַם לֹא לְפִי הַמְחִיר לְבַד.

ש: וּמָה חֲלֻקָה סְבִירָה בְּמַחְסָן גָדוֹל?

א: בַּמַחְסָן שֶׁל אֶלֶף מַקָטִים: בַּקְבוּצָה אֵיי, כַּחֲמִשִׁים מַקָטִים, כְּלוֹמַר חֲמִשָׁה אֲחוּזִים מֵהַפְּרִיטִים, וְהֵם כִּשְׁמוֹנִים אָחוּז מֵעֵרֶךְ הַצְרִיכָה. בַּקְבוּצָה בִּי, כְּמֵאָה וַחֲמִשִׁים מַקָטִים, חֲמִשָׁה עָשָׂר אָחוּז מֵהַפְּרִיטִים, וְכַחֲמִשָׁה עָשָׂר אָחוּז מֵהָעֵרֶךְ. וּבַקְבוּצָה סִי, כִּשְׁמוֹנֶה מֵאוֹת מַקָטִים, שְׁמוֹנִים אָחוּז מֵהַפְּרִיטִים, אֲבָל רַק כַּחֲמִשָׁה אֲחוּזִים מֵהָעֵרֶךְ.

ש: וּמָה הַמְדִינִיוּת לְכֹל קְבוּצָה?

א: פְּרִיטֵי אֵיי: מַחֲזִיקִים בְּכַמוּיוֹת קְטַנוֹת וּמְדוּדוֹת. אֶת הַסִכּוּן לְחֹסֶר פּוֹתְרִים בְּקֶשֶׁב נִהוּלִי: בַּקָרָה קַפְּדָנִית יוֹתֵר, וּסְפִירוֹת מְלַאי תְכוּפוֹת יוֹתֵר. בִּגְלַל הַמִשְׁקָל הַכַּלְכָּלִי שֶׁלָהֶם, הַחִסָכוֹן כָּאן גָדוֹל.
א: פְּרִיטֵי בִּי: מַזְמִינִים בְּדֶרֶךְ כְּלָל כְּמוֹ בַּקְבוּצָה אֵיי, אֲבָל עִם מְלַאי בִּטָחוֹן גָדוֹל יוֹתֵר.
א: וּפְרִיטֵי סִי: מַזְמִינִים בְּכַמוּיוֹת גְדוֹלוֹת מְאוֹד, כְּדֵי שֶׁרֹב הַפְּרִיטִים בַּמַחְסָן יִהְיוּ זְמִינִים. הַמַצֶגֶת קוֹרֵאת לָזֶה שֶׁקֶט נִהוּלִי. כָּךְ אַנְשֵׁי הַמְלַאי יְכוֹלִים לְהַשְׁקִיעַ אֶת רֹב הַמַאֲמָץ בִּפְרִיטֵי אֵיי.

ש: וּמָה עִם הַפִּלוּחִים הָאֲחֵרִים?

א: יֵשׁ שִׁיטָה פְּשׁוּטָה לִזְכֹּר. נַחֲזִיק יוֹתֵר מְלַאי מַפְרִיטִים קְרִיטִיִים, מַפְרִיטִים עִם צְרִיכָה גְבוֹהָה, מַפְרִיטִים עִם זְמַנֵי אַסְפָּקָה אֲרֻכִּים, וּמִפְּרִיטִים שֶׁהָאַסְפָּקָה שֶׁלָהֶם מֻרְכֶּבֶת. וְנַחֲזִיק פָּחוֹת מְלַאי מַפְרִיטִים שֶׁרְגִישִׁים לְפָג תֹקֶף, וּמִפְּרִיטִים שֶׁתוֹפְסִים נֶפַח גָדוֹל.

ש: וּמָה קוֹרֶה כְּשֶׁשְׁנֵי פִּלוּחִים סוֹתְרִים?

א: לְמָשָׁל פְּרִיט אֵיי, שֶׁהוּא גַם קְרִיטִי. אוֹ פְּרִיט סִי, שֶׁרָגִישׁ לְפָג תֹקֶף. הַמַצֶגֶת מַשְׁאִירָה אֶת זֶה כִּשְׁאֵלָה, וְהַתְשׁוּבָה הִיא שִׁקוּל דַעַת: לְשַׁלֵב בֵּין הַשִׁקוּלִים, וְלֹא לִפְעֹל אוֹטוֹמָטִית לְפִי פִּלוּחַ אֶחָד.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: פָּארֶטוֹ מָצָא שֶׁשְׁמוֹנִים אָחוּז מֵהַהוֹן בִּידֵי עֶשְׂרִים אָחוּז מֵהַמִשְׁפָּחוֹת. הַאִם נוֹבֵעַ מִזֶה שֶׁתִשְׁעִים אָחוּז מֵהַהוֹן בִּידֵי עֲשָׂרָה אֲחוּזִים?

~4

א: לֹא. שְׁנֵי הַמִסְפָּרִים מִתְיַחֲסִים לִבְסִיסִים שׁוֹנִים, וְאֵין חוֹבָה שֶׁיַשְׁלִימוּ לְמֵאָה.

ש: שְׁנִיָה: מָה סֵדֶר שְׁלַבֵּי הָעֲבוֹדָה בְּנִתוּחַ פָּארֶטוֹ?

~5

א: מַצִיגִים אֶת הַפְּרִיטִים וְעֶרְכֵיהֶם, מְדָרְגִים מֵהַגָבוֹהַ לְנָמוּךְ, מְחַשְׁבִים אָחוּז יַחֲסִי וּמִצְטַבֵּר, וּמְשַׂרְטְטִים גְרָף וּמַסְקָנוֹת.

ש: שְׁלִישִׁית: סְנִיף אֶחָד קִבֵּל שִׁשִׁים תְלוּנוֹת עַל שְׁלֹשֶׁת אֲלָפִים מְבַקְרִים, וּסְנִיף אַחֵר אַרְבָּעִים וְחָמֵשׁ תְלוּנוֹת עַל תְשַׁע מֵאוֹת מְבַקְרִים. מִי בְּעָיָתִי יוֹתֵר?

~6

א: הַשֵׁנִי. חֲמִשָׁה אֲחוּזִים מֵהַמְבַקְרִים שֶׁלוֹ מִתְלוֹנְנִים, לְעֻמַת שְׁנֵי אֲחוּזִים בָּרִאשׁוֹן.

ש: רְבִיעִית: לְפִי מָה מְדָרְגִים פְּרִיטִים בְּנִתוּחַ אֵיי בִּי סִי?

~4

א: לְפִי עֵרֶךְ הַצְרִיכָה הַשְׁנָתִית: הַצְרִיכָה כָּפוּל מְחִיר הַיְחִידָה.

ש: חֲמִישִׁית: בְּאֵלוּ פְּרִיטִים נַחֲזִיק פָּחוֹת מְלַאי?

~4

א: בִּפְרִיטִים שֶׁרְגִישִׁים לְפָג תֹקֶף, וּבִפְרִיטִים בַּעֲלֵי נֶפַח גָבוֹהַ.

ש: זֶהוּ לַפֶּרֶק הַזֶה. בַּפֶּרֶק הַבָּא: אַחְזָקָה וַאֲמִינוּת. גִישׁוֹת הָאַחְזָקָה, מָה עוֹלָה תַקָלָה, וְאֵיךְ מְחַשְׁבִים אֲמִינוּת שֶׁל מַעֲרֶכֶת בְּטוּר וּבְמַקְבִּיל.
