# title: מלאי ביטחון ונקודת הזמנה
# מצגת 5 של ניצן. כל עובדה כאן מופיעה בבנק 15 או בדף הנוסחאות (כולל שיעורי הבית של שקף 30, פתורים).
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק שְׁלִישִׁי: מְלַאי בִּטָחוֹן וּנְקֻדַת הַזְמָנָה. בַּפֶּרֶק הַקוֹדֵם הִנַחְנוּ שֶׁאֵין חוֹסָרִים. עַכְשָׁו מוֹסִיפִים אֶת חֹסֶר הַוַדָאוּת.

א: וּלְפִי הַמַצֶגֶת, יֵשׁ שְׁנֵי סוּגִים שֶׁלוֹ. מִצַד הַלָקוֹחַ: שִׁנוּיִים בְּקֶצֶב הַצְרִיכָה, שֶׁנִמְדָדִים בִּסְטִיַת הַתֶקֶן שֶׁל הַצְרִיכָה. וּמִצַד הַסַפָּק: שִׁנוּיִים בִּזְמַנֵי הָאַסְפָּקָה, שֶׁנִמְדָדִים בִּסְטִיַת הַתֶקֶן שֶׁל זְמַנֵי הָאַסְפָּקָה. וְהַמִשְׁתַנֶה שֶׁמְשַׁלֵב אֶת שְׁנֵיהֶם הוּא סְטִיַת הַתֶקֶן שֶׁל הַצְרִיכָה בִּזְמַן הָאַסְפָּקָה.

ש: יֵשׁ בְּמַצֶגֶת שְׁנֵי סַפָּקִים לְדֻגְמָה.

א: סַפָּק אֶחָד הִבְטִיחַ שִׁבְעָה יָמִים, וְסִפֵּק בְּפֹעַל בֵּין יוֹמַיִם לְעֶשְׂרִים יָמִים. הַסַפָּק הַשֵׁנִי הִבְטִיחַ עֲשָׂרָה יָמִים, וְסִפֵּק בֵּין תִשְׁעָה לְאֶחָד עָשָׂר. עִם מִי עָדִיף לַעֲבֹד? עִם הַשֵׁנִי. מָה שֶׁקוֹבֵעַ הוּא הַפִּזוּר, לֹא הַהַבְטָחָה. וּזְמַנֵי אַסְפָּקָה יַצִיבִים דוֹרְשִׁים פָּחוֹת מְלַאי בִּטָחוֹן.
א: אַגַב, אֵצֶל הַסַפָּק הָרִאשׁוֹן הַמְמֻצָע בִּכְלָל יָצָא יוֹתֵר מֵעֲשָׂרָה יָמִים. וְאַסְפָּקָה מְהִירָה בִּמְיֻחָד לֹא מְפַצָה עַל הָאִחוּרִים. הִיא רַק מַגְדִילָה אֶת הַפִּזוּר.

ש: וְאוֹתוֹ רַעֲיוֹן לְגַבֵּי מוּצָרִים.

א: כֵּן. מוּצָר שֶׁהַצְרִיכָה הַחָדְשִׁית שֶׁלוֹ קוֹפֶצֶת בֵּין שֶׁבַע לְאַרְבָּעִים וְחָמֵשׁ, צָרִיךְ יוֹתֵר מְלַאי בִּטָחוֹן מִמוּצָר שֶׁנָע בֵּין תְשַׁע עֶשְׂרֵה לְעֶשְׂרִים וּשְׁתַיִם. גַם אִם הַמְמֻצָע שֶׁל הַשֵׁנִי קְצָת גָבוֹהַ יוֹתֵר.

ש: אֵיךְ מְחַשְׁבִים מְלַאי בִּטָחוֹן?

א: הַמַצֶגֶת מַצִיגָה שָׁלוֹשׁ גִישׁוֹת. הַגִישָׁה הַסְטָטִיסְטִית, לְפִי הַהִתְפַּלְגוּת הַנוֹרְמָלִית. גִישָׁה שֶׁנִגְזֶרֶת מִקֶצֶב הַצְרִיכָה, בְּשָׂפָה שֶׁל אַנְשֵׁי הַלוֹגִיסְטִיקָה, לְמָשָׁל חֹדֶשׁ צְרִיכָה. וְגִישָׁה שֶׁנִגְזֶרֶת מֵהַצְרִיכָה בִּזְמַן הָאַסְפָּקָה, לְמָשָׁל חֲמִשִׁים אָחוּז מִמֶנָה.

ש: וְהַגִישָׁה הַסְטָטִיסְטִית הִיא הָעִקָר.

א: נָכוֹן. מְלַאי הַבִּטָחוֹן שָׁוֶה קָיי, כָּפוּל סְטִיַת הַתֶקֶן, כָּפוּל שֹׁרֶשׁ שֶׁל זְמַן הָאַסְפָּקָה.
א: קָיי נִקְבַּע לְפִי רָמַת הַשֵׁרוּת. הַטַבְלָה שֶׁל נִצָן: לִשְׁמוֹנִים וַחֲמִשָׁה אָחוּז, אַחַת נְקֻדָה אַרְבָּעִים וְאַרְבַּע. לִשְׁמוֹנִים וּשְׁמוֹנֶה, אַחַת נְקֻדָה חֲמִשִׁים וָשֵׁשׁ. לְתִשְׁעִים, אַחַת נְקֻדָה שִׁשִׁים וְחָמֵשׁ. לְתִשְׁעִים וּשְׁתַיִם, אַחַת נְקֻדָה שִׁבְעִים וְחָמֵשׁ. וּלְתִשְׁעִים וַחֲמִשָׁה אָחוּז, אַחַת נְקֻדָה תִשְׁעִים וְשֵׁשׁ.

ש: וְיֵשׁ כָּאן שְׁתֵי נְקֻדוֹת עֲדִינוֹת.

א: הָרִאשׁוֹנָה: סְטִיַת הַתֶקֶן הִיא מִדְגָמִית. מְחַלְקִים בְּמִסְפַּר הַתַצְפִּיוֹת פָּחוֹת אַחַת, כְּמוֹ בְּכֹל הַדֻגְמָאוֹת בְּמַצֶגֶת.
א: הַשְׁנִיָה, וְהִיא הֲכִי חֲשׁוּבָה: זְמַן הָאַסְפָּקָה נִמְדָד בְּאוֹתָה יְחִידַת זְמַן שֶׁל סְטִיַת הַתֶקֶן. אִם הַצְרִיכָה שְׁבוּעִית, זְמַן הָאַסְפָּקָה בַּשָׁבוּעוֹת. אִם חָדְשִׁית, בֶּחֳדָשִׁים. וּבַקוּרְס, בַּחֹדֶשׁ יֵשׁ אַרְבָּעָה נְקֻדָה שְׁלוֹשָׁה שָׁבוּעוֹת, אוֹ שְׁלוֹשִׁים יָמִים.

ש: בּוֹא נַעֲבֹר עַל הַדֻגְמָה הָרִאשׁוֹנָה.

א: צְרִיכָה שְׁבוּעִית שֶׁל מֵאָה וְעֶשְׂרִים, מֵאָה וְאַרְבָּעִים, שְׁמוֹנִים, שְׁמוֹנִים, תִשְׁעִים, וּמֵאָה וְעֶשֶׂר. סְטִיַת הַתֶקֶן יוֹצֵאת עֶשְׂרִים וְאַרְבַּע נְקֻדָה עֶשְׂרִים וּשְׁתַיִם. זְמַן הָאַסְפָּקָה אַרְבָּעָה יָמִים, כְּלוֹמַר אַרְבַּע שְׁבִיעִיוֹת שֶׁל שָׁבוּעַ. וְרָמַת שֵׁרוּת שֶׁל תִשְׁעִים וּשְׁתַיִם אָחוּז.
א: אַחַת נְקֻדָה שִׁבְעִים וְחָמֵשׁ, כָּפוּל עֶשְׂרִים וְאַרְבַּע נְקֻדָה עֶשְׂרִים וּשְׁתַיִם, כָּפוּל שֹׁרֶשׁ שֶׁל אַרְבַּע שְׁבִיעִיוֹת. יוֹצֵא בְּעֵרֶךְ שְׁלוֹשִׁים וּשְׁתַיִם יְחִידוֹת.

ש: וּבַשֶׁקֶף כָּתוּב עֶשְׂרִים וְאַרְבַּע.

א: כֵּן, וְזוֹ טָעוּת בַּשֶׁקֶף. עֶשְׂרִים וְאַרְבַּע יוֹצֵא אִם שׁוֹכְחִים אֶת הַשֹׁרֶשׁ. עִם הַשֹׁרֶשׁ, כְּמוֹ בַּנֻסְחָה, שְׁלוֹשִׁים וּשְׁתַיִם. וְיֵשׁ עוֹד טָעוּת, בַּשֶׁקֶף חֲמֵשׁ עֶשְׂרֵה: סְטִיַת הַתֶקֶן שֶׁל מוּצָר אָלֶף הִיא שְׁלוֹשׁ מֵאוֹת שִׁשִׁים וְחָמֵשׁ נְקֻדָה שֵׁשׁ, וְלֹא שְׁלוֹשׁ מֵאוֹת חֲמִשִׁים וְשֵׁשׁ נְקֻדָה שֵׁשׁ. הַסְפָרוֹת הִתְחַלְפוּ.

ש: וְלָמָה בִּכְלָל הַהִתְפַּלְגוּת הַנוֹרְמָלִית?

א: כִּי בְּכֹל תוֹפָעָה שֶׁמִתְפַּלֶגֶת נוֹרְמָלִית, בִּטְוָח שֶׁל סְטִיַת תֶקֶן אַחַת מִכֹּל צַד שֶׁל הַמְמֻצָע נִמְצָאִים בְּעֵרֶךְ שִׁשִׁים וְשִׁבְעָה אָחוּז מֵהַתַצְפִּיוֹת. בִּשְׁתֵי סְטִיוֹת תֶקֶן, כְּתִשְׁעִים וַחֲמִשָׁה אָחוּז. וּבְשָׁלוֹשׁ, תִשְׁעִים וְתִשְׁעָה נְקֻדָה שְׁמוֹנָה אָחוּז.
א: הַדֻגְמָה מֵהַמַצֶגֶת: מִבְחָן עִם מְמֻצָע שִׁבְעִים וָשֵׁשׁ וּסְטִיַת תֶקֶן שֶׁבַע. בֵּין שִׁשִׁים וּשְׁתַיִם לְתִשְׁעִים, כְּלוֹמַר שְׁתֵי סְטִיוֹת תֶקֶן לְכֹל צַד, נִמְצָאִים כְּתִשְׁעִים וַחֲמִשָׁה אָחוּז מֵהַסְטוּדֶנְטִים.

ש: עוֹבְרִים לִנְקֻדַת הַהַזְמָנָה. מָה הַהַגְדָרָה?

א: רָמַת הַמְלַאי שֶׁבָּה מְבַצְעִים אֶת הַהַזְמָנָה. שְׁתֵי מִלִים חֲשׁוּבוֹת. רָמַת מְלַאי: כַּמוּת בַּמְלַאי, וְלֹא נְקֻדַת זְמַן. וּבִצוּעַ הַהַזְמָנָה: לִפְנוֹת לַסַפָּק, וְלֹא קַבָּלַת הַסְחוֹרָה בְּפֹעַל. הַמַרְצֶה קָרָא לָזֶה אַזְעָקָה: כְּשֶׁהַמְלַאי יוֹרֵד לָרָמָה הַזֹאת, מַזְמִינִים.

ש: וְאֵיךְ מְחַשְׁבִים?

א: תָלוּי אִם זְמַן הָאַסְפָּקָה קָצָר מִזְמַן הַמַחְזוֹר, אוֹ אָרֹךְ מִמֶנוּ. כְּשֶׁהוּא קָצָר: מְלַאי הַבִּטָחוֹן, וְעוֹד הַכַּמוּת הַמֻזְמֶנֶת, כָּפוּל זְמַן הָאַסְפָּקָה חֶלְקֵי זְמַן הַמַחְזוֹר.
א: הַדֻגְמָה: מְלַאי בִּטָחוֹן שְׁלוֹשִׁים, כַּמוּת הַזְמָנָה מֵאָה, זְמַן מַחְזוֹר עֶשְׂרִים וְאַרְבָּעָה יָמִים, וּזְמַן אַסְפָּקָה שְׁמוֹנָה עָשָׂר. שְׁלוֹשִׁים, וְעוֹד מֵאָה כָּפוּל שְׁמוֹנֶה עָשָׂר חֶלְקֵי עֶשְׂרִים וְאַרְבַּע. יוֹצֵא מֵאָה וְחָמֵשׁ.

ש: כְּלוֹמַר כְּשֶׁהַמְלַאי יוֹרֵד לְמֵאָה וְחָמֵשׁ יְחִידוֹת, מִתְקַשְׁרִים לַסַפָּק.

א: בְּדִיוּק. וּבְמֶשֶׁךְ שְׁמוֹנָה עָשָׂר יְמֵי הָאַסְפָּקָה נִצְרָכוֹת עוֹד שִׁבְעִים וְחָמֵשׁ יְחִידוֹת, וְהַהַזְמָנָה מַגִיעָה בְּדִיוּק כְּשֶׁנִשְׁאָרוֹת שְׁלוֹשִׁים, מְלַאי הַבִּטָחוֹן. בַּמַצָב הַזֶה נְקֻדַת הַהַזְמָנָה תָמִיד בֵּין מְלַאי הַבִּטָחוֹן לְבֵין מְלַאי הַבִּטָחוֹן וְעוֹד הַכַּמוּת הַמֻזְמֶנֶת.

ש: וּמָה קוֹרֶה כְּשֶׁזְמַן הָאַסְפָּקָה אָרֹךְ מִזְמַן הַמַחְזוֹר?

א: אָז יֵשׁ יוֹתֵר מֵהַזְמָנָה אַחַת בַּדֶרֶךְ. הַנֻסְחָה: מְלַאי הַבִּטָחוֹן, וְעוֹד הַצְרִיכָה הַשְׁנָתִית כָּפוּל זְמַן הָאַסְפָּקָה בַּשָׁנִים, פָּחוֹת הַכַּמוּת הַמֻזְמֶנֶת כָּפוּל מִסְפָּר שָׁלֵם שֶׁל מַחְזוֹרִים. וְהַמִסְפָּר הַשָׁלֵם הוּא זְמַן הָאַסְפָּקָה חֶלְקֵי זְמַן הַמַחְזוֹר, מְעֻגָל כְּלַפֵּי מַטָה.
א: הַדֻגְמָה: מֵפִיץ הַתְרוּפוֹת מֵהַפֶּרֶק הַקוֹדֵם. מְלַאי בִּטָחוֹן שֶׁל חֹדֶשׁ צְרִיכָה, כְּלוֹמַר אֶלֶף. זְמַן מַחְזוֹר שְׁתֵים עֶשְׂרֵה נְקֻדָה שִׁבְעִים וּשְׁתַיִם יָמִים, וּזְמַן אַסְפָּקָה שְׁלוֹשִׁים יָמִים.
א: שְׁלוֹשִׁים חֶלְקֵי שְׁתֵים עֶשְׂרֵה נְקֻדָה שִׁבְעִים וּשְׁתַיִם זֶה שְׁתַיִם נְקֻדָה שְׁלוֹשִׁים וָשֵׁשׁ, וּמַעְגָלִים כְּלַפֵּי מַטָה לִשְׁתַיִם. אֶלֶף, וְעוֹד שְׁנֵים עָשָׂר אֶלֶף כָּפוּל חֹדֶשׁ, כְּלוֹמַר תְשַׁע מֵאוֹת תִשְׁעִים וָשֵׁשׁ, פָּחוֹת אַרְבַּע מֵאוֹת וּשְׁמוֹנֶה עֶשְׂרֵה כָּפוּל שְׁתַיִם. יוֹצֵא אֶלֶף מֵאָה וְשִׁשִׁים.

ש: עַכְשָׁו לִימֵי מַחְסוֹר.

א: יְמֵי מַחְסוֹר בְּשָׁנָה שָׁוִים לְמִסְפַּר הַהַזְמָנוֹת בְּשָׁנָה, כָּפוּל זְמַן הָאַסְפָּקָה בְּיָמִים, כָּפוּל הַהִסְתַבְּרוּת לְחֹסֶר. וְהַהִסְתַבְּרוּת לְחֹסֶר הִיא הַמַשְׁלִים שֶׁל רָמַת הַזְמִינוּת, חֶלְקֵי שְׁתַיִם.

ש: לָמָה חֶלְקֵי שְׁתַיִם?

א: כִּי הַטַבְלָה שֶׁל נִצָן דוּ צְדָדִית. בְּתִשְׁעִים וַחֲמִשָׁה אָחוּז זְמִינוּת, שְׁנֵי וָחֵצִי אָחוּז הֵם הִסְתַבְּרוּת לְעֹדֶף, וּשְׁנֵי וָחֵצִי אָחוּז הִסְתַבְּרוּת לְחֹסֶר.
א: בְּדֻגְמָה: עֶשְׂרִים וּשְׁמוֹנֶה נְקֻדָה שִׁבְעִים וְאַחַת הַזְמָנוֹת, כָּפוּל שְׁלוֹשִׁים יָמִים, כָּפוּל שְׁנֵי וָחֵצִי אָחוּז. יוֹצֵא עֶשְׂרִים וְאֶחָד וָחֵצִי יְמֵי מַחְסוֹר בְּשָׁנָה. וְאִם כֹּל יוֹם חֹסֶר עוֹלֶה אֶלֶף שֶׁקֶל, בְּהֶפְסֵדֵי עִסְקָאוֹת וּבְהֶפְסֵדֵי מוֹנִיטִין, זֶה עֶשְׂרִים וְאֶחָד אֶלֶף וַחֲמֵשׁ מֵאוֹת שֶׁקֶל בְּשָׁנָה.

ש: וְאֵיךְ מִשְׁתַנָה עֲלוּת הָאַחְזָקָה?

א: מְלַאי הַבִּטָחוֹן נִמְצָא בַּמְלַאי כֹּל הַזְמַן, וְלָכֵן הוּא נוֹסָף בִּמְלוֹאוֹ לַמְלַאי הַמְמֻצָע. בְּדֻגְמָה מְנִיחִים שֶׁסְטִיַת הַתֶקֶן הַשְׁבוּעִית הִיא חֲצִי מֵהַצְרִיכָה הַשְׁבוּעִית, כְּלוֹמַר מֵאָה שֵׁשׁ עֶשְׂרֵה. מְלַאי הַבִּטָחוֹן בְּתִשְׁעִים וַחֲמִשָׁה אָחוּז: אַחַת נְקֻדָה תִשְׁעִים וָשֵׁשׁ, כָּפוּל מֵאָה שֵׁשׁ עֶשְׂרֵה, כָּפוּל שֹׁרֶשׁ שֶׁל אַרְבַּע נְקֻדָה שָׁלוֹשׁ. יוֹצֵא אַרְבַּע מֵאוֹת שִׁבְעִים וְאַחַת.
א: וַעֲלוּת הָאַחְזָקָה וְהַהַזְמָנוֹת: אַרְבַּע מֵאוֹת שִׁבְעִים וְאַחַת, וְעוֹד מָאתַיִם וְתֵשַׁע, כָּפוּל אַרְבָּעִים וְאַרְבַּע, וְעוֹד תִשְׁעַת אֲלָפִים מֵאָה שְׁמוֹנִים וְשֶׁבַע. בְּסַךְ הַכֹּל, שְׁלוֹשִׁים וְתִשְׁעָה אֶלֶף מֵאָה וְשֵׁשׁ שֶׁקֶל.

ש: וּבַשֶׁקֶף הָאַחֲרוֹן יֵשׁ שִׁעוּרֵי בַּיִת: מָה קוֹרֶה אִם הַזְמִינוּת יוֹרֶדֶת לְתִשְׁעִים וּשְׁתַיִם, אוֹ לְתִשְׁעִים אָחוּז.

א: פָּתַרְנוּ אוֹתָם בְּדַף הַנֻסְחָאוֹת. מְלַאי הַבִּטָחוֹן יוֹרֵד, לְאַרְבַּע מֵאוֹת עֶשְׂרִים וְאַחַת, וְאַחַר כָּךְ לִשְׁלוֹשׁ מֵאוֹת תִשְׁעִים וְשֶׁבַע. עֲלוּת הָאַחְזָקָה יוֹרֶדֶת בְּכַמָה אַלְפֵי שְׁקָלִים. אֲבָל יְמֵי הַמַחְסוֹר קוֹפְצִים: שְׁלוֹשִׁים וְאַרְבָּעָה וָחֵצִי, וְאַחַר כָּךְ אַרְבָּעִים וּשְׁלוֹשָׁה.
א: כְּשֶׁכֹּל יוֹם חֹסֶר עוֹלֶה אֶלֶף שֶׁקֶל, תִשְׁעִים וַחֲמִשָׁה אָחוּז יוֹצֵא הַזוֹל בְּיוֹתֵר בְּסַךְ הַכֹּל: כְּשִׁשִׁים וְאֶחָד אֶלֶף שֶׁקֶל, לְעֻמַת כְּשִׁבְעִים וְאֶחָד אֶלֶף וְכִשְׂבֵעִים וְתִשְׁעָה אֶלֶף. הַחִסָכוֹן בָּאַחְזָקָה קָטָן מֵהַתוֹסֶפֶת בַּעֲלוּת הַחֹסֶר.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: סְטִיַת תֶקֶן שֶׁל צְרִיכָה חָדְשִׁית, וּזְמַן אַסְפָּקָה שֶׁל שְׁבוּעַיִם. מָה מַצִיבִים בְּתוֹךְ הַשֹׁרֶשׁ?

~5

א: שְׁבוּעַיִם בֶּחֳדָשִׁים: שְׁתַיִם חֶלְקֵי אַרְבַּע נְקֻדָה שָׁלוֹשׁ.

ש: שְׁנִיָה: מָה עֵרֶךְ קָיי לְרָמַת שֵׁרוּת שֶׁל תִשְׁעִים אָחוּז, לְפִי הַטַבְלָה שֶׁל נִצָן?

~4

א: אַחַת נְקֻדָה שִׁשִׁים וְחָמֵשׁ.

ש: שְׁלִישִׁית: מְלַאי בִּטָחוֹן אַרְבָּעִים, כַּמוּת הַזְמָנָה מָאתַיִם וְאַרְבָּעִים, זְמַן מַחְזוֹר שְׁלוֹשִׁים יָמִים, וּזְמַן אַסְפָּקָה עֲשָׂרָה יָמִים. מָה נְקֻדַת הַהַזְמָנָה?

~6

א: מֵאָה וְעֶשְׂרִים. אַרְבָּעִים, וְעוֹד מָאתַיִם וְאַרְבָּעִים כָּפוּל עֶשְׂרֵה חֶלְקֵי שְׁלוֹשִׁים.

ש: רְבִיעִית: רָמַת זְמִינוּת שֶׁל תִשְׁעִים אָחוּז. מָה הַהִסְתַבְּרוּת לְחֹסֶר?

~4

א: חֲמִשָׁה אֲחוּזִים. הַמַשְׁלִים הוּא עֲשָׂרָה אֲחוּזִים, וּמְחַלְקִים בִּשְׁתַיִם.

ש: חֲמִישִׁית: מָה יִקְרֶה אִם מוֹרִידִים אֶת רָמַת הַזְמִינוּת מִתִשְׁעִים וַחֲמִשָׁה לְתִשְׁעִים אָחוּז?

~4

א: מְלַאי הַבִּטָחוֹן וַעֲלוּת הָאַחְזָקָה יִקְטְנוּ, אֲבָל יְמֵי הַמַחְסוֹר וַעֲלוּת הַחֹסֶר יִגְדְלוּ.

ש: זֶהוּ לַפֶּרֶק הַזֶה. בַּפֶּרֶק הַבָּא: פָּארֶטוֹ. לָמָה שְׁמוֹנִים עֶשְׂרִים לֹא חַיָב לְהַשְׁלִים לְמֵאָה, וְאֵיךְ מְחַלְקִים מַחְסָן לְשָׁלוֹשׁ קְבוּצוֹת: אֵיי, בִּי, סִי.
