# title: עדכון לפני המבחן — מבנה המבחן והבחינה לדוגמה של ניצן
# פרק עדכון (7.10.2026), אחרי שיעור הסיכום של ניצן מ-6.10. כל עובדה כאן מהתמלול של אותו שיעור ומהבחינה לדוגמה שהוצגה בו (בנקים 23–24, ראש דף הנוסחאות).
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק עִדְכּוּן. אַחֲרֵי שֶׁהִקְלַטְנוּ אֶת הַפְּרָקִים הַקוֹדְמִים, נִצָן הֶעֱבִיר שִׁעוּר סִכּוּם, וְהִצִיג בּוֹ בְּחִינָה לְדֻגְמָה. מָה הִתְחַדֵשׁ?

א: הַרְבֵּה. קֹדֶם כֹּל, מִבְנֵה הַמִבְחָן. עֶשְׂרִים וְחָמֵשׁ שְׁאֵלוֹת אָמֵרִיקָאִיוֹת: שְׁלוֹשׁ עֶשְׂרֵה שֶׁל נִצָן, וּשְׁתַיִם עֶשְׂרֵה שֶׁל שְׁלֹמֹה.
א: מִתוֹךְ הַשְׁאֵלוֹת שֶׁל נִצָן, בְּעֵרֶךְ חֲצִי כַּמוּתִיוֹת. בַּמִלִים שֶׁלוֹ: יֵשׁ רַק חָמֵשׁ אוֹ שֵׁשׁ שְׁאֵלוֹת מָתֵמָטִיוֹת, וְהֵן פְּשׁוּטוֹת.
א: הַמִבְחָן בְּחֹמֶר פָּתוּחַ. אֲבָל נִצָן לֹא מְחַלֵק דַף נֻסְחָאוֹת. הוּא אָמַר: אֲנִי בָּא בְּלִי חֹמֶר, אַתֶם צְרִיכִים לְהָבִיא אֶת כֹּל הַחֹמֶר, וּלְאַרְגֵן אוֹתוֹ.
א: וּמַחְשְׁבוֹן. כֹּל מַחְשְׁבוֹן מַתְאִים, בִּתְנַאי שֶׁיוֹדְעִים לְהָפִיק מִמֶנוּ רֶגְרֶסְיָה. לֹא מַחְשֵׁב, וְלֹא טֵלֵפוֹן. מִי שֶׁמַחֲזִיק טֵלֵפוֹן, נִפְסָל.

ש: וּמָה לְגַבֵּי הֶקֵף הַחֹמֶר?

א: זֶה הָעִדְכּוּן הֲכִי חָשׁוּב. נִצָן אָמַר: אֶצְלְכֶם זֶה עַד מַצֶגֶת שֶׁבַע. כְּלוֹמַר אִינְטֶגְרַצְיָה, חִזוּי, נִהוּל מְלַאי, מְלַאי בִּטָחוֹן, פָּארֶטוֹ, וְאַחְזָקָה.
א: וּמָה שֶׁאַחֲרֵי מַצֶגֶת שֶׁבַע, לֹא בַּמִבְחָן. זֶה אוֹמֵר שֶׁחִשׁוּבֵי הָאֲמִינוּת, הַטוּר וְהַמַקְבִּיל, שֶׁדִבַּרְנוּ עֲלֵיהֶם בְּפֶרֶק חָמֵשׁ, וְכֹל פֶּרֶק שֵׁשׁ, עַל תְדִירוּת הָאַחְזָקָה הַמוֹנַעַת וְעַל נְקֻדַת הָאִזוּן, הֵם לְהַעֲשָׁרָה בִּלְבַד.

ש: וְיֵשׁ עוֹד דְבָרִים שֶׁיָרְדוּ?

א: כֵּן. הַחְלָקָה מַעֲרִיכִית: לֹא נִלְמְדָה, וְלֹא תִהְיֶה. הַגְזֵרָה שֶׁל נֻסְחַת הַכַּמוּת הַכַּלְכָּלִית: לֹא לְמִבְחָן, אֲבָל הַנֻסְחָה עַצְמָה נִמְצֵאת בְּמַצֶגֶת. חִשׁוּבִים שֶׁל אַחְזָקָה חֲזוּיָה: לֹא יִהְיוּ. וּבְחִזוּי עוֹנָתִי, כְּמוֹ שֶׁכְּבָר אָמַרְנוּ, הַדֵקוֹמְפּוֹזִיצְיָה וְהַשִׁיטָה הַפְּשׁוּטָה בַּחוּץ.

ש: עַכְשָׁיו לַבְּחִינָה לְדֻגְמָה. אֵיךְ הִיא נִרְאֵית?

א: עָשָׂר שְׁאֵלוֹת, עִם חָמֵשׁ תְשׁוּבוֹת לְכֹל שְׁאֵלָה, וְלֹא אַרְבַּע. נִצָן אָמַר שֶׁבַּמִבְחָן עַצְמוֹ הַשְׁאֵלוֹת יִהְיוּ קְצָרוֹת וּפְשׁוּטוֹת יוֹתֵר, כִּי צָרִיךְ גַם זְמַן לַעֲנוֹת.
א: עַל שְׁתַיִם מֵהַשְׁאֵלוֹת הוּא אָמַר שֶׁהֵן לֹא בַּחֹמֶר, וְשֶׁהוּא יוֹרִיד אוֹתָן. אָז נִשְׁאֲרוּ חָמֵשׁ שְׁאֵלוֹת חִשׁוּב, וְשָׁלוֹשׁ שְׁאֵלוֹת הֲבָנָה. בִּשְׁאֵלוֹת הַחִשׁוּב יֵשׁ שִׁשָׁה סוּגִים שֶׁל חִשׁוּב. נַעֲבֹר עֲלֵיהֶם.

ש: הַסוּג הָרִאשׁוֹן.

א: תַחֲזִית לְפִי קַו נָתוּן. מְנַהֵל טוֹעֵן שֶׁמִסְפַּר הַפְּנִיוֹת עוֹלֶה לְפִי קַו יָשָׁר: מֵאָה וַחֲמִשִׁים כָּפוּל אִיקְס, וְעוֹד תִשְׁעַת אֲלָפִים. יֵשׁ נְתוּנִים שֶׁל חָמֵשׁ שָׁנִים, וְרוֹצִים תַחֲזִית לַשָׁנָה הַבָּאָה.
א: הַמַלְכֹּדֶת הַיְחִידָה: אִיקְס הוּא מִסְפַּר הַתְקוּפָה, וְלֹא הַשָׁנָה. חָמֵשׁ שְׁנוֹת נְתוּנִים, אָז הַשָׁנָה הַבָּאָה הִיא תְקוּפָה שֵׁשׁ. מֵאָה וַחֲמִשִׁים כָּפוּל שֵׁשׁ, וְעוֹד תִשְׁעַת אֲלָפִים: תִשְׁעַת אֲלָפִים וּתְשַׁע מֵאוֹת. נִצָן אָמַר: רְאִיתֶם כַּמָה זְמַן זֶה לָקַח? דַקָה.

ש: הַסוּג הַשֵׁנִי.

א: רֶגְרֶסְיָה בְּמַחְשְׁבוֹן. מַקְלִידִים אֶת הַנְתוּנִים, וּמוֹצִיאִים שְׁלוֹשָׁה מִסְפָּרִים: הַחוֹתֵךְ, הַשִׁפּוּעַ, וּמְקַדֵם הַמִתְאָם.
א: בְּטוּר שֶׁל אִיקְס מַקְלִידִים אַחַת, שְׁתַיִם, שָׁלוֹשׁ, אַרְבַּע, חָמֵשׁ. מִסְפָּרִים סִדוּרִיִים, וְלֹא שָׁנִים. בַּטוּר הַשֵׁנִי, אֶת הַנְתוּנִים. וְהַתַחֲזִית הִיא הַחוֹתֵךְ, וְעוֹד הַשִׁפּוּעַ כָּפוּל מִסְפַּר הַתְקוּפָה הַבָּאָה.
א: בְּדֻגְמָה שֶׁל נִצָן יָצָא חוֹתֵךְ שֶׁל שְׁמוֹנַת אֲלָפִים מָאתַיִם שְׁמוֹנִים וְשֶׁבַע, וְשִׁפּוּעַ שֶׁל מֵאָה וְשָׁלוֹשׁ עֶשְׂרֵה. לִתְקוּפָה שֵׁשׁ: שְׁמוֹנַת אֲלָפִים תְשַׁע מֵאוֹת שִׁשִׁים וְחָמֵשׁ.
א: וּשְׁנֵי דְבָרִים שֶׁהוּא הִדְגִישׁ. בְּמַחְשְׁבוֹן, הָאוֹת בִּי הִיא תָמִיד הַשִׁפּוּעַ. וְחַיָבִים לְתַרְגֵל אֶת זֶה בַּבַּיִת, עַל הַמַחְשְׁבוֹן שֶׁלָכֶם, כִּי בְּכֹל דֶגֶם הַמַקָשִׁים קְצָת שׁוֹנִים.

ש: וְאֵיךְ יוֹדְעִים אִם אֶפְשָׁר לִסְמֹךְ עַל הַתַחֲזִית?

א: זֶה הַסוּג הַשְׁלִישִׁי: רֶגְרֶסְיָה, וּשְׁאֵלָה אִם כְּדַאי לְהִשְׁתַמֵשׁ בַּשִׁיטָה. מִסְתַכְּלִים עַל מְקַדֵם הַמִתְאָם. בְּדֻגְמָה שֶׁל הַסָפָארִי הוּא יָצָא אֶפֶס נְקֻדָה תִשְׁעִים וְשָׁלוֹשׁ. מֵעַל תִשְׁעִים אָחוּז, זֶה קֶשֶׁר לִינֵאָרִי חָזָק וְחִיוּבִי, וְלָכֵן כְּדַאי לְהִשְׁתַמֵשׁ בַּשִׁיטָה.
א: בַּתְשׁוּבוֹת הַשְׁגוּיוֹת יוֹפִיעוּ מִלִים כְּמוֹ חַלָשׁ, בֵּינוֹנִי, אוֹ שְׁלִילִי. אָז צָרִיךְ גַם אֶת הַמִסְפָּר הַנָכוֹן, וְגַם אֶת הַתֵאוּר הַנָכוֹן.

ש: הַסוּג הָרְבִיעִי, מֵהַמְלַאי.

א: רָמַת הַמְלַאי הַמַקְסִימָלִית: מְלַאי הַבִּטָחוֹן, וְעוֹד כַּמוּת הַהַזְמָנָה. אֵלֶה שְׁנֵי חִשׁוּבִים נִפְרָדִים, שֶׁמְחַבְּרִים בַּסוֹף.
א: בְּדֻגְמָה: צְרִיכָה שְׁנָתִית שֶׁל אֶלֶף וּמָאתַיִם יְחִידוֹת, וּמַזְמִינִים פַּעַם בְּחֹדֶשׁ. אָז כַּמוּת הַהַזְמָנָה הִיא אֶלֶף וּמָאתַיִם חֶלְקֵי שְׁתֵים עֶשְׂרֵה: מֵאָה.
א: וּמְלַאי הַבִּטָחוֹן: רָמַת שֵׁרוּת שֶׁל תִשְׁעִים אָחוּז, אָז הַמְקַדֵם הוּא אַחַת נְקֻדָה שִׁשִׁים וְחָמֵשׁ. סְטִיַת הַתֶקֶן הַשְׁבוּעִית שְׁמוֹנִים, וּזְמַן הָאַסְפָּקָה שָׁבוּעַ אֶחָד. אַחַת נְקֻדָה שִׁשִׁים וְחָמֵשׁ, כָּפוּל שְׁמוֹנִים, כָּפוּל שֹׁרֶשׁ שֶׁל אַחַת: מֵאָה שְׁלוֹשִׁים וּשְׁתַיִם. וְיַחַד: מָאתַיִם שְׁלוֹשִׁים וּשְׁתַיִם.

ש: וְאֵלוּ רָמוֹת שֵׁרוּת יְכוֹלוֹת לְהוֹפִיעַ?

א: רַק אַרְבַּע, מֵהַטַבְלָה שֶׁבַּמַצֶגֶת. שְׁמוֹנִים וּשְׁמוֹנֶה אָחוּז: אַחַת נְקֻדָה חֲמִשִׁים וָשֵׁשׁ. תִשְׁעִים: אַחַת נְקֻדָה שִׁשִׁים וְחָמֵשׁ. תִשְׁעִים וּשְׁתַיִם: אַחַת נְקֻדָה שִׁבְעִים וְחָמֵשׁ. וְתִשְׁעִים וַחֲמִשָׁה: אַחַת נְקֻדָה תִשְׁעִים וָשֵׁשׁ. הַטַבְלָה לֹא מְחֻלֶקֶת בַּמִבְחָן, אָז הִיא צְרִיכָה לִהְיוֹת בַּחֹמֶר שֶׁמְבִיאִים.

ש: וְהַסוּג הַחֲמִישִׁי, מֵהָאַחְזָקָה.

א: עֲלוּת אַחְזָקָה שְׁנָתִית, וּזְמַן הַשְׁבָּתָה. נִצָן אָמַר שֶׁכֹּל הָעֲלוּיוֹת הָאֵלֶה הֵן מִשְׂחָק שֶׁל שְׁלוֹשָׁה מַרְכִּיבִים קְבוּעִים: הַשְׁבָּתָה, שְׂכַר טֶכְנַאי, וְחוֹמָרִים.
א: מְחַשְׁבִים תַקָלַת שֶׁבֶר אַחַת. בְּדֻגְמָה, שְׁתֵים עֶשְׂרֵה שְׁעוֹת הַשְׁבָּתָה, כָּפוּל אֶלֶף שְׁקָלִים. וְעוֹד שְׂכַר הַטֶכְנָאִים: שְׁעָתַיִם שֶׁל טֶכְנַאי אֶחָד בְּזִהוּי, וְעֶשֶׂר שָׁעוֹת שֶׁל שְׁנֵי טֶכְנָאִים בְּתִקוּן, כְּלוֹמַר עֶשְׂרִים וּשְׁתַיִם שְׁעוֹת עֲבוֹדָה, כָּפוּל מֵאָה שְׁקָלִים. וְעוֹד חֲלָקִים, אֶלֶף וְאַרְבַּע מֵאוֹת. יַחַד: חֲמִשָׁה עָשָׂר אֶלֶף וְשֵׁשׁ מֵאוֹת.
א: וְאָז טִפּוּל מוֹנֵעַ אֶחָד: שָׁעָה וָחֵצִי שֶׁל הַשְׁבָּתָה, שָׁעָה וָחֵצִי שֶׁל טֶכְנַאי, וְחוֹמָרִים. יַחַד: אֶלֶף וּשְׁמוֹנֶה מֵאוֹת. וְשִׂימוּ לֵב: גַם טִפּוּל מוֹנֵעַ שֶׁנַעֲשָׂה בִּזְמַן הָעֲבוֹדָה מַשְׁבִּית אֶת הַצִיוּד.
א: וּבַשָׁנָה: תַקָלָה אַחַת, וְשִׁשָׁה טִפּוּלִים, אֶחָד בְּכֹל חוֹדְשַׁיִם. חֲמִשָׁה עָשָׂר אֶלֶף וְשֵׁשׁ מֵאוֹת, וְעוֹד שֵׁשׁ כָּפוּל אֶלֶף וּשְׁמוֹנֶה מֵאוֹת: עֶשְׂרִים וְשִׁשָׁה אֶלֶף וְאַרְבַּע מֵאוֹת שְׁקָלִים. וְהַהַשְׁבָּתָה: שְׁתֵים עֶשְׂרֵה שָׁעוֹת, וְעוֹד שֵׁשׁ כָּפוּל שָׁעָה וָחֵצִי: עֶשְׂרִים וְאַחַת שָׁעוֹת.

ש: זֶה נִשְׁמָע אָרֹךְ.

א: נָכוֹן. וְנִצָן אָמַר שֶׁבַּמִבְחָן, שְׁאֵלָה כָּזֹאת תִהְיֶה פְּשׁוּטָה יוֹתֵר. לְמָשָׁל, בְּלִי הַחֲלֻקָה לְטֶכְנַאי אֶחָד וְלִשְׁנֵי טֶכְנָאִים.

ש: וְהַסוּג הַשִׁשִׁי?

א: מְמֻצָע נָע מְשֻׁקְלָל. בִּבְחִינָה לְדֻגְמָה הוּא מְנֻסָח עִם אֲחוּז הַפְחָתָה. אֲחוּז הַפְחָתָה שֶׁל שְׁמוֹנָה אֲחוּזִים פֵּרוּשׁוֹ: לַשָׁנָה הָאַחֲרוֹנָה מִשְׁקָל אַחַת, וְלַשָׁנָה שֶׁלְפָנֶיהָ אֶפֶס נְקֻדָה תִשְׁעִים וּשְׁתַיִם. מַכְפִּילִים, מְחַבְּרִים, וּמְחַלְקִים בִּסְכוּם הַמִשְׁקָלוֹת: אַחַת נְקֻדָה תִשְׁעִים וּשְׁתַיִם.
א: וּבְאוֹתוֹ אֹפֶן, עֲשָׂרָה אֲחוּזִים הֵם אַחַת, אֶפֶס נְקֻדָה תֵשַׁע, אֶפֶס נְקֻדָה שְׁמוֹנָה. וְעֶשְׂרִים אֲחוּזִים: אַחַת, אֶפֶס נְקֻדָה שְׁמוֹנָה, אֶפֶס נְקֻדָה שֵׁשׁ.

ש: וּמָתַי בּוֹחֲרִים אֲחוּז הַפְחָתָה גָבוֹהַ?

א: זוֹ אַחַת מִשְׁאֵלוֹת הַהֲבָנָה. אֲחוּז הַפְחָתָה גָבוֹהַ נוֹתֵן לַשָׁנָה הָאַחֲרוֹנָה אֶת רֹב הַמִשְׁקָל. זֶה מַתְאִים לְמוּצָר טֶכְנוֹלוֹגִי שֶׁמִשְׁתַנֶה מַהֵר, כִּי מָה שֶׁהָיָה לִפְנֵי כַּמָה שָׁנִים כְּבָר פָּחוֹת רֵלֵוַנְטִי. לְמוּצָר בְּסִיסִי וְיַצִיב, כְּמוֹ קוֹטֵג', נוֹתְנִים אֲחוּז הַפְחָתָה נָמוּךְ, וְלוֹקְחִים יוֹתֵר שָׁנִים.

ש: וּשְׁאֵלוֹת הַהֲבָנָה הָאֲחֵרוֹת?

א: אַחַת עַל מְלַאי הַבִּטָחוֹן: הוּא מִתְחַשֵׁב בִּסְטִיַת הַתֶקֶן שֶׁל הַצְרִיכָה בִּזְמַן הָאַסְפָּקָה, וּבְרָמַת הַשֵׁרוּת. וְהוּא לֹא עוֹלֶה בְּאֹפֶן לִינֵאָרִי עִם זְמַן הָאַסְפָּקָה, כִּי זְמַן הָאַסְפָּקָה נִמְצָא בְּתוֹךְ הַשֹׁרֶשׁ.
א: וְהַשְׁנִיָה עַל פָּארֶטוֹ, וְהִיא מְנֻסַחַת בִּשְׁלִילָה: אֵיזֶה מִשְׁפָּט אֵינוֹ נָכוֹן. פְּרִיט מִקְבוּצָה אֵיי, שֶׁגַם רָגִישׁ לִתְפוּגָה וְגַם תוֹפֵס נֶפַח, מַחֲזִיקִים בְּכַמוּת קְטַנָה, וּבְבַקָרָה קַפְּדָנִית. פְּרִיט מִקְבוּצָה סִי, שֶׁהוּא קְרִיטִי וְלֹא מִתְיַשֵׁן, מַחְזִיקִים הַרְבֵּה.
א: וְנִצָן הִבְטִיחַ: לֹא יִהְיוּ שְׁאֵלוֹת עִם קוֹנְפְלִיקְטִים. אוֹ שֶׁכֹּל הַשִׁקוּלִים אוֹמְרִים לְהַחְזִיק יוֹתֵר, אוֹ שֶׁכֻּלָם אוֹמְרִים לְהַחְזִיק פָּחוֹת.

ש: וְיֵשׁ עוֹד מַשֶׁהוּ בַּבְּחִינָה לְדֻגְמָה?

א: בְּסוֹף הַמִסְמָךְ צֹרְפוּ שֶׁבַע שְׁאֵלוֹת הֲבָנָה קְצָרוֹת: עַל פִּתְרוֹן הָאִינְטֶגְרַצְיָה, עַל שׁוֹט הַשׁוֹר, עַל אַחְזָקַת שֶׁבֶר, עַל מַדַד הַזְמִינוּת, עַל הַגִישָׁה הַנָאִיבִית, וְעַל בְּחִירַת מוֹדֵל הַחִזוּי לְפִי הַסְטִיָה הַקְטַנָה בְּיוֹתֵר. כֻּלָן נִמְצָאוֹת בַּאֲתַר הַתִרְגוּל.

ש: וְעַכְשָׁיו שְׁאֵלָה לַדֶרֶךְ. הַצְרִיכָה הַשְׁנָתִית הִיא אַלְפַּיִם וְאַרְבַּע מֵאוֹת יְחִידוֹת, וּמַזְמִינִים פַּעַם בְּחֹדֶשׁ. רָמַת הַשֵׁרוּת תִשְׁעִים וַחֲמִשָׁה אָחוּז, סְטִיַת הַתֶקֶן הַשְׁבוּעִית חֲמִשִׁים, וּזְמַן הָאַסְפָּקָה שָׁבוּעַ. מַהִי רָמַת הַמְלַאי הַמַקְסִימָלִית?
~6
א: כַּמוּת הַהַזְמָנָה: אַלְפַּיִם וְאַרְבַּע מֵאוֹת חֶלְקֵי שְׁתֵים עֶשְׂרֵה, מָאתַיִם. מְלַאי הַבִּטָחוֹן: אַחַת נְקֻדָה תִשְׁעִים וָשֵׁשׁ כָּפוּל חֲמִשִׁים, תִשְׁעִים וּשְׁמוֹנֶה. וְיַחַד: מָאתַיִם תִשְׁעִים וּשְׁמוֹנֶה.

ש: וְעוֹד אַחַת. יֵשׁ נְתוּנִים שֶׁל חָמֵשׁ שָׁנִים, וְהַקַו הוּא שְׁלוֹשׁ מֵאוֹת, וְעוֹד עֶשְׂרִים כָּפוּל אִיקְס. מָה הַתַחֲזִית לַשָׁנָה הַבָּאָה?
~5
א: הַשָׁנָה הַבָּאָה הִיא תְקוּפָה שֵׁשׁ. שְׁלוֹשׁ מֵאוֹת, וְעוֹד עֶשְׂרִים כָּפוּל שֵׁשׁ: אַרְבַּע מֵאוֹת וְעֶשְׂרִים.

ש: וְאֵיךְ מִתְרַגְלִים אֶת כֹּל זֶה?

א: בַּאֲתַר הַתִרְגוּל יֵשׁ עַכְשָׁיו שְׁנֵי הֶקֵפִים חֲדָשִׁים. הָרִאשׁוֹן, הַבְּחִינָה לְדֻגְמָה שֶׁל נִצָן, עִם הַפִּתְרוֹן הַמָלֵא, וְעִם מָה שֶׁהוּא אָמַר עַל כֹּל שְׁאֵלָה. וְהַשֵׁנִי, שְׁאֵלוֹת בְּסִגְנוֹן הַבְּחִינָה לְדֻגְמָה: אוֹתָם סוּגֵי חִשׁוּב, עִם מִסְפָּרִים אֲחֵרִים.
א: סִימוּלַצְיַת הַמִבְחָן בְּנוּיָה עַכְשָׁיו כְּמוֹ הַמִבְחָן עַצְמוֹ: שְׁלוֹשׁ עֶשְׂרֵה שְׁאֵלוֹת שֶׁל נִצָן, שֵׁשׁ מֵהֶן חִשׁוּב, וּשְׁתַיִם עֶשְׂרֵה שֶׁל שְׁלֹמֹה. וּבְרֹאשׁ דַף הַנֻסְחָאוֹת נִמְצָא כֹּל מָה שֶׁנֶאֱמַר עַל הַמִבְחָן, עִם הַפִּתְרוֹנוֹת שֶׁל הַבְּחִינָה לְדֻגְמָה.

ש: אָז בְּהַצְלָחָה בַּמִבְחָן.

א: בְּהַצְלָחָה.
