# title: שלמה — אסטרטגיה, זמישות וארגז הכלים
# מצגות שא 7 ו-שא 8 של שלמה ארליך. כל עובדה כאן מופיעה בבנקים 7 ו-8 או בדף הנוסחאות.
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.
# מונחים באנגלית נאמרים בעברית ככל האפשר (רזה, זמישה, דחיפה, משיכה).

ש: פֶּרֶק עֲשִׂירִי, וְהָאַחֲרוֹן שֶׁל שְׁלֹמֹה: אַסְטְרָטֶגְיָה, שַׁרְשֶׁרֶת אַסְפָּקָה זְמִישָׁה, וְאַרְגַז הַכֵּלִים שֶׁל הַמְנַהֵל. נַתְחִיל מֵהַהַגְדָרָה.

א: שַׁרְשֶׁרֶת הָאַסְפָּקָה כּוֹלֶלֶת אֶת כֹּל הַשְׁלַבִּים, מֵחוֹמְרֵי הַגֶלֶם וְעַד הַלָקוֹחַ. זוֹרְמִים בָּה חֹמֶר, מֵידָע, וְכֶסֶף. וּשְׁלֹמֹה מַדְגִישׁ: הַלָקוֹחַ הוּא חֵלֶק מֵהַשַׁרְשֶׁרֶת. הִיא לֹא נִגְמֶרֶת אֵצֶל הַקִמְעוֹנַאי.
א: וְעִם הַזְמַן, הַשַׁרְשֶׁרֶת הוֹפֶכֶת לְרֶשֶׁת: תַהֲלִיכִים מַקְבִּילִים, וּמִשְׁלוֹחַ יָשִׁיר מֵהַסַפָּקִים לְלָקוֹחוֹת, בְּלִי לַעֲבֹר דֶרֶךְ מַחְסָן מֶרְכָּזִי.

ש: וּמָה זֶה דְחִיפָה וּמְשִׁיכָה?

א: דְחִיפָה: יִצוּר לַתַחֲזִית וְלִמְלַאי, לִפְנֵי שֶׁיֵשׁ הַזְמָנָה. מְשִׁיכָה: פְּעִילוּת לְפִי הַזְמָנַת הַלָקוֹחַ. וּנְקֻדַת הַנִתוּק בֵּינֵיהֶן הִיא קַבָּלַת הַזְמָנַת הַלָקוֹחַ.
א: דֻגְמָה: רֶשֶׁת רָהִיטִים מְיַצֶרֶת שִׁלְדֵי סַפּוֹת סְטַנְדַרְטִיִים לְפִי תַחֲזִית, וּמְרַפֶּדֶת בְּבַד שֶׁהַלָקוֹחַ בָּחַר רַק אַחֲרֵי הַהַזְמָנָה. הַשְׁלָדִים הֵם דְחִיפָה, הָרִפּוּד הוּא מְשִׁיכָה.

ש: וּמָה הַמַטָרוֹת שֶׁל שַׁרְשֶׁרֶת הָאַסְפָּקָה?

א: חָמֵשׁ. יְצִירַת עֵרֶךְ מוּסָף לָאִרְגוּן. הָעֵרֶךְ שֶׁל הַשַׁרְשֶׁרֶת. הָרִוְחִיוּת שֶׁל הַשַׁרְשֶׁרֶת, שֶׁהִיא הַהֶפְרֵשׁ בֵּין הַהַכְנָסוֹת מֵהַלָקוֹחַ לְבֵין סַךְ הָעֲלוּיוֹת שֶׁל הַשַׁרְשֶׁרֶת. הַצָעַת הָעֵרֶךְ שֶׁל הַשַׁרְשֶׁרֶת. וּמַעֲבָר מִמֶרְכַּז עֲלוּת לְמֶרְכַּז רֶוַח.
א: חִשׁוּב קָטָן: הַכְנָסוֹת שֶׁל שְׁנֵים עָשָׂר מִילְיוֹן, וַעֲלוּיוֹת שֶׁל כֹּל הַחֻלְיוֹת תִשְׁעָה מִילְיוֹן. רִוְחִיוּת הַשַׁרְשֶׁרֶת: שְׁלוֹשָׁה מִילְיוֹן.

ש: עַכְשָׁו לִזְמִישׁוּת.

א: שְׁלֹמֹה מְפָרֵק אֶת הַמִלָה לִשְׁלוֹשָׁה: זְמִינוּת, זְרִיזוּת, וּגְמִישׁוּת. וְהַהַגְדָרָה: יְכֹלֶת הַתְאָמָה מְהִירָה לִנְסִבּוֹת מִשְׁתַנוֹת, וְלַצְעָדִים שֶׁל הַמִתְחָרִים. יַצְרָנִית מַשְׁקָאוֹת שֶׁזִהֲתָה בְּתוֹךְ שְׁבוּעַיִם שֶׁמִתְחָרֶה הִשִׁיק טַעַם מַצְלִיחַ, וְשִׁנְתָה אֶת תֹכְנִית הַיִצוּר? זוֹ זְמִישׁוּת.

ש: וְאֵיךְ בּוֹחֲרִים אַסְטְרָטֶגְיָה לְשַׁרְשֶׁרֶת?

א: הִיא נִגְזֶרֶת מֵהָאַסְטְרָטֶגְיָה שֶׁל הָאִרְגוּן, שֶׁמִתְחָרֶה בָּאֵיכוּת, בַּזְמַן, בִּגְמִישׁוּת, אוֹ בַּעֲלוּת. מִמֶנָה נִגְזֶרֶת אַסְטְרָטֶגְיַת הַשַׁרְשֶׁרֶת: שַׁרְשֶׁרֶת יְעִילָה, שֶׁמַדְגִישָׁה עֲלוּת, אוֹ שַׁרְשֶׁרֶת תְגוּבָתִית, שֶׁמַדְגִישָׁה אֵפֶקְטִיבִיוּת. וּמֵהֶן, הַיְכוֹלוֹת וְהַבִּצוּעִים.
א: אִרְגוּן שֶׁמִתְחָרֶה עַל זְמַן אַסְפָּקָה קָצָר וְעַל זְמַן יְצִיאָה מָהִיר לַשׁוּק, בּוֹחֵר בְּאַסְטְרָטֶגְיַת זְמַן. מִמֶנָה נִגְזֶרֶת שַׁרְשֶׁרֶת תְגוּבָתִית.

ש: וְזֶה מֵבִיא אוֹתָנוּ לַהַתְאָמָה הָאַסְטְרָטֵגִית.

א: הַכְּלָל: בִּקוּשׁ יָדוּעַ וְיַצִיב, שַׁרְשֶׁרֶת יְעִילָה. בִּקוּשׁ לֹא וַדָאִי, שַׁרְשֶׁרֶת תְגוּבָתִית. מֶלַח שֻׁלְחָנִי? שַׁרְשֶׁרֶת יְעִילָה. מַכְשִׁיר חַדְשָׁנִי בְּלִי נְתוּנֵי עָבָר? שַׁרְשֶׁרֶת תְגוּבָתִית.
א: וְהַמַלְכֹּדֶת: יַצְרָנִית נְיַר טוּאָלֵט, עִם בִּקוּשׁ יַצִיב, שֶׁבָּנְתָה קִבֹּלֶת גְמִישָׁה, מְלַאי בִּטָחוֹן וְסַפָּקִים יְקָרִים. זוֹ חֹסֶר הַתְאָמָה. הִיא מְשַׁלֶמֶת עַל תְגוּבָתִיוֹת שֶׁהִיא לֹא צְרִיכָה.

ש: וְאֵיךְ מוֹדְדִים אִי וַדָאוּת?

א: יֵשׁ טַבְלָה. בְּאִי וַדָאוּת נְמוּכָה: אִי דִיוּק בְּתַחֲזִית שֶׁל כַּעֲשָׂרָה אֲחוּזִים, חוֹסָרִים שֶׁל אֶחָד עַד שְׁנַיִם אֲחוּזִים, וּבְלִי עוֹנָתִיוֹת. וּבְאִי וַדָאוּת גְבוֹהָה: אִי דִיוּק שֶׁל אַרְבָּעִים עַד מֵאָה אָחוּז, חוֹסָרִים שֶׁל עֲשָׂרָה עַד אַרְבָּעִים אָחוּז, וְעוֹנָתִיוֹת שֶׁל עֲשָׂרָה עַד עֶשְׂרִים וַחֲמִשָׁה אָחוּז. וְשִׂימוּ לֵב: אִי הַוַדָאוּת הַגְבוֹהָה בָּאָה עִם רִוְחִיוּת גְבוֹהָה.

ש: וּמִי צָרִיךְ שַׁרְשֶׁרֶת זְמִישָׁה?

א: שְׁלֹמֹה מְחַלֵק אֶת הַמוּצָרִים לְפִי פָּארֶטוֹ. עֶשְׂרִים אָחוּז מֵהַמוּצָרִים, שֶׁהֵם שְׁמוֹנִים אָחוּז מֵהַבִּקוּשׁ, מְנֹהָלִים בִּרְזָה: יִצוּר לְתַחֲזִית, מְלַאי מֶרְכָּזִי, וְיִתְרוֹנוֹת לְגֹדֶל. וּשְׁמוֹנִים אָחוּז מֵהַמוּצָרִים, שֶׁהֵם רַק עֶשְׂרִים אָחוּז מֵהַבִּקוּשׁ, מְנֹהָלִים בִּזְמִישׁוּת: יִצוּר לְהַזְמָנָה, וּתְגוּבָה מְהִירָה.
א: וְהַמִשְׁפָּט שֶׁכְּדַאי לִזְכֹּר: תַחֲזִית לְקִבֹּלֶת, בִּצוּעַ לְפִי הַבִּקוּשׁ. כְּלוֹמַר, הַתַחֲזִית קוֹבַעַת כַּמָה קִבֹּלֶת לְהָכִין, וְהַיִצוּר עַצְמוֹ לְפִי הַבִּקוּשׁ בְּפֹעַל.

ש: וְיֵשׁ מַטְרִיצָה.

א: שְׁנֵי צִירִים: הַאִם הַבִּקוּשׁ צָפוּי, וְהַאִם זְמַן הָאַסְפָּקָה קָצָר אוֹ אָרֹךְ. בִּקוּשׁ צָפוּי וּזְמַן אַסְפָּקָה אָרֹךְ: לְתַכְנֵן וּלְבַצֵעַ, זֶה רָזֶה. בִּקוּשׁ צָפוּי וּזְמַן אַסְפָּקָה קָצָר: ג'ַאסְט אִין טַיים. בִּקוּשׁ לֹא צָפוּי וּזְמַן אַסְפָּקָה אָרֹךְ: לְגַדֵר בְּאֶמְצָעוּת מְלַאי. וּבִקוּשׁ לֹא צָפוּי וּזְמַן אַסְפָּקָה קָצָר: לְהָגִיב וּלְבַצֵעַ, זֶה זָמִישׁ.
א: לְמָשָׁל, יְבוּאָן צַעֲצוּעִים לַחַגִים, עִם זְמַן אַסְפָּקָה שֶׁל אַרְבָּעָה חוֹדָשִׁים מֵהַמִזְרָח הָרָחוֹק, וּבִקוּשׁ שֶׁאִי אֶפְשָׁר לְצַפּוֹת לְכֹל דֶגֶם. הוּא לֹא יָכוֹל לְהָגִיב מַהֵר, וְלָכֵן מְגַדֵר עִם מְלַאי.

ש: וּמָה הַהֶבְדֵל בֵּין רָזֶה לְזָמִישׁ?

א: רָזָה: יְעִילוּת וַעֲלוּת, מִקוּד פְּנִימָה בָּאִרְגוּן, שַׁרְשֶׁרֶת אַסְפָּקָה, דְחִיפָה, וּמְכִירָה מֵהַמְלַאי. זָמִישׁ: אֵפֶקְטִיבִיוּת וּתְגוּבָתִיוֹת, מִקוּד בַּלָקוֹחַ, שַׁרְשֶׁרֶת בִּקוּשׁ, מְשִׁיכָה, וּמָנָה שֶׁל יְחִידָה אַחַת. יַצְרָנִית רִהוּט שֶׁמְיַצֶרֶת כֹּל פְּרִיט לְפִי הַזְמָנָה, בְּמִדוֹת אִישִׁיוֹת? זְמִישָׁה.
א: וּלְשַׁרְשֶׁרֶת בִּקוּשׁ זְמִישָׁה יֵשׁ אַרְבָּעָה מְאַפְיְנִים. רְגִישׁוּת לַשׁוּק. וִירְטוּאָלִיוֹת: שַׁרְשֶׁרֶת שֶׁמְבֻסֶסֶת עַל מֵידָע מֵהַשׁוּק, וְלֹא עַל הַמְלַאי. לְמָשָׁל רֶשֶׁת שֶׁמְשַׁדֶרֶת לַסַפָּקִים כֹּל יוֹם אֶת נְתוּנֵי הַקֻפּוֹת. שִׁלוּב תַהֲלִיכִים בֵּין סַפָּקִים לְלָקוֹחוֹת, כְּמוֹ פִּתוּחַ מְשֻׁתָף. וּבָסִיס שֶׁל רֶשֶׁת.

ש: וְיֵשׁ טַבְלָה שֶׁמַשְׁוָה בֵּין שְׁתֵי הָאַסְטְרָטֶגְיוֹת.

א: כֵּן, שׁוּרָה אַחֲרֵי שׁוּרָה. הַיִעוּד: עֲלוּת נְמוּכָה, מוּל תְגוּבָה מְהִירָה. תִיכוֹן הַמוּצָר: עֲלוּת מִינִימָלִית, מוּל מוּצָר מוֹדוּלָרִי שֶׁאֶפְשָׁר לְהַתְאִים. הַמְחָרָה: רִוְחִיוּת נְמוּכָה, מוּל רִוְחִיוּת גְבוֹהָה. יִצוּר: נְצִילוּת גְבוֹהָה, מוּל קִבֹּלֶת גְמִישָׁה. מְלַאי: מִינִימָלִי, מוּל מְלַאי בִּטָחוֹן. זְמַנֵי אַסְפָּקָה: לְקַצֵר בַּעֲלוּת מֵיטָבִית, מוּל לְקַצֵר בְּכֹל מְחִיר. סַפָּקִים: לְפִי עֲלוּת, מוּל סַפָּקִים זְמִישִׁים וְאֵיכוּתִיִים. וַהֲפָצָה: זוֹלָה, מוּל מְהִירָה.
א: הַמַלְכֹּדֶת בְּשׁוּרַת הַהַמְחָרָה: הָרִוְחִיוּת הַגְבוֹהָה שַׁיֶכֶת לַשַׁרְשֶׁרֶת הַתְגוּבָתִית, וְלֹא לִיעִילָה.

ש: וְהַמִקְרֶה שֶׁל קַארְדוֹ?

א: בִּשְׁנַת אַלְפַּיִם וַחֲמֵשׁ עֶשְׂרֵה, הָיוּ יוֹתֵר מִמָאתַיִם מוּצָרִים סוֹפִיִים, כֹּל אֶחָד מֻתְאָם לַמֵפִיץ וְלַמְדִינָה, יִצוּר לְהַזְמָנָה, וְשִׁשָׁה שָׁבוּעוֹת הוֹבָלָה יַמִית מֵהַמִזְרָח. אַחַר כָּךְ מֵעֵבֶר לְסַפָּקִים מְקוֹמִיִים, מְסִירָה מֵהוּנְגַרְיָה, וְשִׁשָׁה יְמֵי הוֹבָלָה יַבַּשְׁתִית. וּבַסוֹף, מוּצָר אֶחָד, וְיִצוּר לַמְלַאי. הַפְחָתַת הַשׁוֹנוּת הִיא שֶׁאִפְשְׁרָה לְיַצֵר לַמְלַאי, לְקַצֵר זְמַנִים וּלְהוֹרִיד מְלַאי.

ש: עוֹבְרִים לַמַצֶגֶת הָאַחֲרוֹנָה, אַרְגַז הַכֵּלִים. מָה נוֹסָף לָאֶתְגָר?

א: אֵיכוּת. בַּמָבוֹא הָאֶתְגָר הָיָה לְהַבְטִיחַ אַסְפָּקוֹת תוֹךְ זְמִישׁוּת וּמְלַאי נָמוּךְ. כָּאן: לְהַבְטִיחַ אַסְפָּקוֹת לְפִי תֹכְנִית הַמְכִירוֹת, תוֹךְ גְמִישׁוּת תִפְעוּלִית, מְלַאי נָמוּךְ, וְאֵיכוּת גְבוֹהָה.

ש: וְיֵשׁ מוֹדֵל שֶׁל הַשָׁנָה הָרִאשׁוֹנָה בַּתַפְקִיד.

א: קֹדֶם כֹּל, עֲקֻמַת שְׂבִיעוּת הָרָצוֹן. הֶלֶם בַּהַתְחָלָה. שִׂיא קָטָן אַחֲרֵי שָׁבוּעַ. נְקֻדַת שֵׁפֶל בִּסְבִיבוֹת הַחֹדֶשׁ. הִתְאוֹשְׁשׁוּת עַד שְׁלוֹשָׁה חוֹדָשִׁים, וַעֲלִיָה עַד שָׁנָה.
א: וְאָז אַרְבָּעָה רִבְעוֹנִים. בָּרִבְעוֹן הָרִאשׁוֹן: לְהַקְשִׁיב, לִלְמֹד, לֹא לְבַטֵל אֶת הֶעָבָר, וּלְהַכִּיר אֶת הָאִרְגוּן וְאֶת סְבִיבַת הַפְּעֻלָה. בָּרִבְעוֹן הַשֵׁנִי: תִכְנוּן. תֹכְנִית עֲבוֹדָה עִם יִעוּד, אַסְטְרָטֶגְיָה, קוֹדֵי הִתְנַהֲגוּת וּמַדָדִים. בָּרִבְעוֹן הַשְׁלִישִׁי: יִשׁוּם הַשִׁנוּי, תְחִלַת הַחְלָפַת כּוֹחַ אָדָם, וְנִהוּל סַפָּקִים. וּבָרִבְעוֹן הָרְבִיעִי: בַּקָרָה וּלְמִידָה, וְתֹכְנִית לַשָׁנָה הַבָּאָה.

ש: אֵיפֹה קַל לְהִתְבַּלְבֵּל?

א: בָּרִבְעוֹן שֶׁל הַחְלָפַת כּוֹחַ הָאָדָם. זֶה לֹא הָרִבְעוֹן הָרִאשׁוֹן, שֶׁבּוֹ מַקְשִׁיבִים, וְגַם לֹא הַשֵׁנִי, שֶׁבּוֹ מְתַכְנְנִים. זֶה הַשְׁלִישִׁי.
א: וְעוֹד שְׁתֵי הַבְחָנוֹת מִתֹכְנִית הָעֲבוֹדָה. הַיִעוּד שֶׁל הַשַׁרְשֶׁרֶת נִגְזַר מֵהֶחָזוֹן שֶׁל הָאִרְגוּן, וְקוֹדֵי הַהִתְנַהֲגוּת נִגְזָרִים מֵהָעֲרָכִים שֶׁלוֹ.

ש: וְאֵיךְ מְנַהֲלִים וּמוֹדְדִים?

א: עֵץ מְדָדִים, עִם כַּרְטִיס צִיוּן מְאֻזָן, וּמַדָדִים לְפִי מוֹדֵל סְמַארְט. לוּחַ שְׁעוֹנִים אִרְגוּנִי, שֶׁמְבֻסָס עַל נְתוּנִים זְמִינִים וַאֲמִינִים. דוּחַ שְׁבוּעִי: שִׁתוּף מֵידָע, וְתִכְנוּן מוּל בִּצוּעַ. דוּחַ חוֹדְשִׁי: הַמַדָדִים הָעִקָרִיִים. בַּקָרָה רִבְעוֹנִית: מְדָדִים וּמַעֲקָב מְשִׂימוֹת. וְתֹכְנִית שְׁנָתִית: תִקוּף הַיִעוּד, הָאַסְטְרָטֶגְיָה וְהַתַקְצִיב.

ש: וּמָה עִם תְפִיסַת הַהַפְעָלָה?

א: שְׁנֵי מַצָבִים. כְּשֶׁשַׁרְשֶׁרֶת הָאַסְפָּקָה הִיא לֹא הַלִבָּה, הִיא מַרְכִּיב הוֹצָאָה מַשְׁמָעוּתִי, וּבוֹחֲנִים לָה מִקוּר חוּץ. וּכְשֶׁהִיא הַלִבָּה, הִיא חֵלֶק מֵהַצָעַת הָעֵרֶךְ, הַמַרְכִּיב הָעִקָרִי בַּהוֹצָאוֹת, וְיֵשׁ לָה פּוֹטֶנְצְיָאל לִהְיוֹת מֶרְכַּז רֶוַח. אָז בּוֹחֲנִים לְהַרְחִיב שֵׁרוּתִים, וְלִמְכֹּר אוֹתָם גַם לְאִרְגוּנִים אֲחֵרִים. וּבִשְׁנֵי הַמַצָבִים, הִיא קְרִיטִית לַיִעוּד שֶׁל הָאִרְגוּן.

ש: וְנִשְׁאֲרוּ הַמוֹדֵלִים הַמִתְקַדְמִים.

א: שְׁלוֹשָׁה עִקָרִיִים. הַשַׁרְשֶׁרֶת הַשִׁתוּפִית פּוֹתֶרֶת חֹסֶר אִזוּן: מַשְׁאַבִּים בְּטֵלִים בְּמָקוֹם אֶחָד, וְחֹסֶר בְּמָקוֹם אַחֵר. דֶרֶךְ פְּלַטְפוֹרְמָה מְשַׁתְפִים מַשְׁאַבִּים, עוֹבְרִים מֵהוֹצָאוֹת קְבוּעוֹת לְמִשְׁתַנוֹת, וּמַגְדִילִים נְצִילוּת.
א: הַשַׁרְשֶׁרֶת הַשְׁטוּחָה עוֹבֶרֶת מִנִהוּל שַׁרְשֶׁרֶת הָאַסְפָּקָה לְנִהוּל שַׁרְשֶׁרֶת הַבִּקוּשׁ. מָה שֶׁאִפְשֵׁר אוֹתָה הוּא הַטֶכְנוֹלוֹגְיָה: מְהִירוּת הַתִקְשֹׁרֶת, וּזְמִינוּת הַמֵידָע. וְהָרְכִיבִים שֶׁלָה: תַחֲזִית אֲמִינָה, מְלַאי זָמִין, זְמַנֵי אַסְפָּקָה קְצָרִים, צִמְצוּם מִגְוַן הַמוּצָרִים, דַחְיָנוּת, אִחוּד מֶרְכְּזֵי הֲפָצָה, וּמִשְׁלוֹחִים קְטַנִים וּתְכוּפִים. וְשִׂימוּ לֵב: צִמְצוּם הַמִגְוָן, וְלֹא הַגְדָלָה.
א: וְהַשַׁרְשֶׁרֶת הַהִתְנַהֲגוּתִית עוֹסֶקֶת בְּהַטָיוֹת. מְנַהֵל שֶׁמְסָרֵב לִמְכֹּר מְלַאי יָשָׁן מִתַחַת לַעֲלוֹת, כִּי אָסוּר לְמַמֵשׁ הֶפְסֵד: שִׂנְאַת הֶפְסֵד. קַנְיָן שֶׁמַזְמִין כַּמוּת כְּפוּלָה בִּגְלַל מִבְצָע אֶחָד וְעוֹד אֶחָד: הַטָיַת הַמִבְצָעִים.

ש: וּמִי אַחְרַאי עַל הַסַפָּקִים?

א: מַחְזוֹר חַיֵי הַסַפָּק. הַבְּחִירָה, אֵצֶל הָרֶכֶשׁ. הַנִהוּל הַשׁוֹטֵף, אֵצֶל שַׁרְשֶׁרֶת הָאַסְפָּקָה. וְהַהַעֲרָכָה הַתְקוּפָתִית וְהָאֵרוּעִים הַחֲרִיגִים, אֵצֶל נִהוּל הָאֵיכוּת.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: הַכְנָסוֹת שֶׁל עֶשְׂרִים מִילְיוֹן, וַעֲלוּיוֹת שַׁרְשֶׁרֶת שֶׁל חֲמִשָׁה עָשָׂר מִילְיוֹן. מָה רִוְחִיוּת הַשַׁרְשֶׁרֶת?

~5

א: חֲמִשָׁה מִילְיוֹן.

ש: שְׁנִיָה: בִּקוּשׁ לֹא צָפוּי, וּזְמַן אַסְפָּקָה אָרֹךְ. מָה עוֹשִׂים לְפִי הַמַטְרִיצָה?

~4

א: מְגַדְרִים בְּאֶמְצָעוּת מְלַאי.

ש: שְׁלִישִׁית: בְּאֵיזֶה רִבְעוֹן בַּשָׁנָה הָרִאשׁוֹנָה מַתְחִילִים לְהַחֲלִיף כּוֹחַ אָדָם?

~4

א: בָּרִבְעוֹן הַשְׁלִישִׁי.

ש: רְבִיעִית: לְאֵיזוֹ שַׁרְשֶׁרֶת שַׁיֶכֶת רִוְחִיוּת גְבוֹהָה בְּטַבְלַת הַהַשְׁוָאָה?

~4

א: לַשַׁרְשֶׁרֶת הַתְגוּבָתִית.

ש: חֲמִישִׁית: מָה מְשֻׁתָף לִשְׁנֵי הַמַצָבִים בִּתְפִיסַת הַהַפְעָלָה?

~4

א: בִּשְׁנֵי הַמַצָבִים, הַשַׁרְשֶׁרֶת קְרִיטִית לַיִעוּד שֶׁל הָאִרְגוּן.

ש: זֶהוּ לַפֶּרֶק הַזֶה, וְזֶה סוֹגֵר אֶת הַחֹמֶר שֶׁל שְׁנֵי הַמַרְצִים. בַּפֶּרֶק הָאַחֲרוֹן: חֲזָרָה לְמִבְחָן. מַלְכּוֹדוֹת הַחִשׁוּב, הַטָעוּיוֹת בַּשְׁקָפִים, וּמָה אָמְרוּ הַמַרְצִים עַל הַחֹמֶר לַמִבְחָן.
